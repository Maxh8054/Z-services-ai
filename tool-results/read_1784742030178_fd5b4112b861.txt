    52→function sanitizeForFilename(text: string): string {
    53→  return text
    54→    .trim()
    55→    .substring(0, 60)
    56→    .replace(/[<>:"/\\|?*\x00-\x1f]/g, '')
    57→    .replace(/\s+/g, '_')
    58→    .replace(/_+/g, '_')
    59→    .replace(/^_|_$/g, '')
    60→    || 'export';
    61→}

    63→function generateReportId(): string {
    64→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No I, O, 0, 1 to avoid confusion
    65→  let id = 'ZS-';
    66→  for (let i = 0; i < 6; i++) {
    67→    id += chars.charAt(Math.floor(Math.random() * chars.length));
    68→  }
    69→  return id;
    70→}

   101→// Componente de Menu de Ações Mobile
   102→function MobileActionMenu({
   103→  onAddPhoto,
   104→  onDataDialog,
   105→  onExportDialog,
   106→  onEmailDialog,
   107→  onClearConfirm,
   108→  reportData,
   109→  reportType,
   110→  t,
   111→}: {
   112→  onAddPhoto: () => void;
   113→  onDataDialog: () => void;
   114→  onExportDialog: () => void;
   115→  onEmailDialog: () => void;
   116→  onClearConfirm: () => void;
   117→  reportData: unknown;
   118→  reportType: 'home' | 'inspecao';
   119→  t: (key: string, params?: Record<string, string | number>) => string;
   120→}) {
   121→  const [isOpen, setIsOpen] = useState(false);
   122→  const [timeLeft, setTimeLeft] = useState(10);
   123→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   124→  const countdownRef = useRef(10);
   125→
   126→  useEffect(() => {
   127→    if (isOpen) {
   128→      // Reset countdown ref
   129→      countdownRef.current = 10;
   130→      
   131→      timerRef.current = setInterval(() => {
   132→        countdownRef.current -= 1;
   133→        const newTime = countdownRef.current;
   134→        setTimeLeft(newTime);
   135→        
   136→        if (newTime <= 0) {
   137→          setIsOpen(false);
   138→          if (timerRef.current) {
   139→            clearInterval(timerRef.current);
   140→            timerRef.current = null;
   141→          }
   142→        }
   143→      }, 1000);
   144→      
   145→      return () => {
   146→        if (timerRef.current) {
   147→          clearInterval(timerRef.current);
   148→          timerRef.current = null;
   149→        }
   150→      };
   151→    }
   152→  }, [isOpen]);
   153→
   154→  const handleAction = (action: () => void) => {
   155→    action();
   156→    setIsOpen(false);
   157→    if (timerRef.current) {
   158→      clearInterval(timerRef.current);
   159→      timerRef.current = null;
   160→    }
   161→  };
   162→
   163→  return (
   164→    <div className="md:hidden fixed bottom-4 right-4 z-50">
   165→      {/* Menu expandido */}
   166→      {isOpen && (
   167→        <div className="absolute bottom-16 right-0 bg-black/95 backdrop-blur-sm rounded-2xl p-3 shadow-2xl border border-white/10 min-w-[200px]">
   168→          <div className="text-xs text-white/60 mb-2 text-center">
   169→            Fechando em {timeLeft}s
   170→          </div>
   171→          <div className="grid grid-cols-3 gap-2">
   172→            <Button 
   173→              size="icon" 
   174→              className="rounded-full bg-green-600 hover:bg-green-700 h-12 w-12" 
   175→              onClick={() => handleAction(onAddPhoto)}
   176→              title={t('action.addPhoto')}
   177→            >
   178→              <Plus className="h-5 w-5" />
   179→            </Button>
   180→            <Button 
   181→              size="icon" 
   182→              className="rounded-full bg-blue-600 hover:bg-blue-700 h-12 w-12" 
   183→              onClick={() => handleAction(onDataDialog)}
   184→              title={t('action.importExport')}
   185→            >
   186→              <FileDown className="h-5 w-5" />
   187→            </Button>
   188→            <Button 
   189→              size="icon" 
   190→              className="rounded-full bg-purple-600 hover:bg-purple-700 h-12 w-12" 
   191→              onClick={() => handleAction(onExportDialog)}
   192→              title={t('action.download')}
   193→            >
   194→              <Download className="h-5 w-5" />
   195→            </Button>
   196→            <Button 
   197→              size="icon" 
   198→              className="rounded-full bg-orange-500 hover:bg-orange-600 h-12 w-12" 
   199→              onClick={() => handleAction(onEmailDialog)}
   200→              title="Email"
   201→            >
   202→              <Mail className="h-5 w-5" />
   203→            </Button>
   204→            <Button 
   205→              size="icon" 
   206→              className="rounded-full bg-red-600 hover:bg-red-700 h-12 w-12" 
   207→              onClick={() => handleAction(onClearConfirm)}
   208→              title={t('action.clear')}
   209→            >
   210→              <Trash2 className="h-5 w-5" />
   211→            </Button>
   212→          </div>
   213→        </div>
   214→      )}
   215→      
   216→      {/* Botão principal */}
   217→      <Button
   218→        className={`rounded-full h-14 w-14 shadow-lg ${isOpen ? 'bg-red-500 hover:bg-red-600' : 'bg-orange-500 hover:bg-orange-600'}`}
   219→        onClick={() => setIsOpen(!isOpen)}
   220→      >
   221→        {isOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   222→      </Button>
   223→    </div>
   224→  );
   225→}

   369→// Componente para a seção de peças adicionais
   370→function AdditionalPartsSection({
   371→  parentPn,
   372→  photoId,
   373→  additionalParts,
   374→  onAddPart,
   375→  onRemovePart,
   376→  onUpdatePart,
   377→  t,
   378→}: {
   379→  parentPn: string;
   380→  photoId?: string;
   381→  additionalParts: AdditionalPart[];
   382→  onAddPart: (part: AdditionalPart) => void;
   383→  onRemovePart: (id: string) => void;
   384→  onUpdatePart?: (id: string, data: Partial<AdditionalPart>) => void;
   385→  t: (key: string, params?: Record<string, string | number>) => string;
   386→}) {
   387→  const [newPn, setNewPn] = useState('');
   388→  const [newSerialNumber, setNewSerialNumber] = useState('');
   389→  const [newPartName, setNewPartName] = useState('');
   390→  const [newQuantity, setNewQuantity] = useState('');
   391→  const [newCriticality, setNewCriticality] = useState<'Alta' | 'Média' | 'Baixa' | ''>('');
   392→  const [showExcelPaste, setShowExcelPaste] = useState(false);
   393→  const [excelPasteData, setExcelPasteData] = useState('');
   394→
   395→  const photoAdditionalParts = additionalParts.filter(p =>
   396→    (photoId && p.photoId === photoId) || (!photoId && p.parentPn === parentPn)
   397→  );
   398→
   399→  const handleAddPart = () => {
   400→    if (!newPn.trim() || !newPartName.trim() || !newQuantity.trim()) return;
   401→    
   402→    onAddPart({
   403→      id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   404→      pn: newPn.trim(),
   405→      serialNumber: newSerialNumber.trim(),
   406→      partName: newPartName.trim(),
   407→      quantity: newQuantity.trim(),
   408→      criticality: newCriticality,
   409→      parentPn: parentPn,
   410→      photoId: photoId || undefined,
   411→    });
   412→
   413→    setNewPn('');
   414→    setNewSerialNumber('');
   415→    setNewPartName('');
   416→    setNewQuantity('');
   417→    setNewCriticality('');
   418→  };
   419→
   420→  // Processar dados do Excel colados
   421→  const handlePasteExcel = () => {
   422→    if (!excelPasteData.trim()) return;
   423→    
   424→    // Parse dos dados - formato: Part No.      Q'ty    Part name (tab ou espaços)
   425→    const lines = excelPasteData.trim().split('\n');
   426→    
   427→    lines.forEach((line) => {
   428→      // Divide por tab ou múltiplos espaços
   429→      const parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p);
   430→      
   431→      if (parts.length >= 3) {
   432→        const pn = parts[0];
   433→        const quantity = parts[1];
   434→        const partName = parts.slice(2).join(' '); // Nome pode ter espaços
   435→        
   436→        onAddPart({
   437→          id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   438→          pn: pn,
   439→          serialNumber: '',
   440→          partName: partName,
   441→          quantity: quantity,
   442→          criticality: '',
   443→          parentPn: parentPn,
   444→          photoId: photoId || undefined,
   445→        });
   446→      }
   447→    });
   448→    
   449→    setExcelPasteData('');
   450→    setShowExcelPaste(false);
   451→  };
   452→
   453→  // Get badge color based on criticality
   454→  const getCriticalityBadge = (criticality: string) => {
   455→    switch (criticality) {
   456→      case 'Alta':
   457→        return <Badge className="bg-red-500 text-white text-xs h-5 px-2">{t('photo.criticalityHigh')}</Badge>;
   458→      case 'Média':
   459→        return <Badge className="bg-yellow-500 text-white text-xs h-5 px-2">{t('photo.criticalityMedium')}</Badge>;
   460→      case 'Baixa':
   461→        return <Badge className="bg-green-500 text-white text-xs h-5 px-2">{t('photo.criticalityLow')}</Badge>;
   462→      default:
   463→        return null;
   464→    }
   465→  };
   466→
   467→  return (
   468→    <div className="mt-3 p-3 bg-orange-50 rounded-lg border-2 border-orange-400 space-y-2">
   469→      <div className="flex items-center justify-between">
   470→        <Label className="text-sm text-orange-800 font-bold flex items-center gap-2">
   471→          <ListTree className="h-4 w-4" />
   472→          {t('subparts.partsOf', { pn: parentPn || '...' })}
   473→        </Label>
   474→        <Button
   475→          variant="ghost"
   476→          size="sm"
   477→          className="h-7 text-xs text-orange-600 hover:text-orange-800"
   478→          onClick={() => setShowExcelPaste(!showExcelPaste)}
   479→        >
   480→          {showExcelPaste ? 'Manual' : 'Colar Excel'}
   481→        </Button>
   482→      </div>
   483→      
   484→      {/* Excel Paste Mode */}
   485→      {showExcelPaste && (
   486→        <div className="space-y-2">
   487→          <p className="text-xs text-orange-700">
   488→            Cole os dados do Excel no formato: Part No. | Q'ty | Part name (uma linha por peça)
   489→          </p>
   490→          <Textarea
   491→            className="min-h-[80px] text-sm"
   492→            placeholder="4244836&#9;4&#9;BUCHA&#10;4244837&#9;2&#9;ANEL"
   493→            value={excelPasteData}
   494→            onChange={(e) => setExcelPasteData(e.target.value)}
   495→          />
   496→          <div className="flex gap-2">
   497→            <Button
   498→              size="sm"
   499→              className="bg-orange-500 hover:bg-orange-600"
   500→              onClick={handlePasteExcel}
   501→              disabled={!excelPasteData.trim()}
   502→            >
   503→              <Plus className="h-4 w-4 mr-1" /> Adicionar
   504→            </Button>
   505→            <Button
   506→              variant="outline"
   507→              size="sm"
   508→              onClick={() => { setExcelPasteData(''); setShowExcelPaste(false); }}
   509→            >
   510→              Cancelar
   511→            </Button>
   512→          </div>
   513→        </div>
   514→      )}
   515→      
   516→      {/* Manual Mode */}
   517→      {!showExcelPaste && (
   518→        <>
   519→      
   520→      {photoAdditionalParts.length > 0 && (
   521→        <div className="space-y-1 max-h-32 overflow-y-auto">
   522→          {photoAdditionalParts.map((part) => (
   523→            <div key={part.id} className="flex items-center gap-2 bg-white p-2 rounded border text-sm">
   524→              <span className="text-orange-600 font-mono font-bold">└─</span>
   525→              <span className="font-semibold text-gray-700">{part.pn}</span>
   526→              {part.serialNumber && (
   527→                <span className="text-gray-500">({part.serialNumber})</span>
   528→              )}
   529→              <span className="text-gray-600 flex-1 truncate">{part.partName}</span>
   530→              <Badge variant="secondary" className="text-xs h-5 px-2">{t('photo.quantity')}: {part.quantity}</Badge>
   531→              {getCriticalityBadge(part.criticality)}
   532→              <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 min-h-[44px] min-w-[44px] md:min-h-0 md:min-w-0" onClick={() => onRemovePart(part.id)}>
   533→                <X className="h-4 w-4" />
   534→              </Button>
   535→            </div>
   536→          ))}
   537→        </div>
   538→      )}
   539→
   540→      {/* Mobile layout - stacked */}
   541→      <div className="md:hidden space-y-2">
   542→        <div className="grid grid-cols-2 gap-2">
   543→          <Input className="h-11 text-base" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />
   544→          <Input className="h-11 text-base" placeholder={t('photo.serialPlaceholder')} value={newSerialNumber} onChange={(e) => setNewSerialNumber(e.target.value)} />
   545→        </div>
   546→        <Input className="h-11 text-base" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   547→        <div className="grid grid-cols-2 gap-2">
   548→          <Input className="h-11 text-base" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />

   578→            <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>
   579→          </SelectContent>
   580→        </Select>
   581→        <Button size="icon" className="h-8 w-8 bg-orange-500 hover:bg-orange-600 col-span-2" onClick={handleAddPart} disabled={!newPn.trim() || !newPartName.trim() || !newQuantity.trim()}>
   582→          <Plus className="h-4 w-4" />
   583→        </Button>
   584→      </div>
   585→        </>
   586→      )}
   587→    </div>
   588→  );
   589→}

   594→// Editor de Foto
   595→function PhotoEditor({
   596→  photo,
   597→  editMode = 'primary',
   598→  onSave,
   599→  onClose,
   600→  t,
   601→}: {
   602→  photo: PhotoData;
   603→  editMode?: PhotoEditMode;
   604→  onSave: (data: Partial<PhotoData>) => void;
   605→  onClose: () => void;
   606→  t: (key: string, params?: Record<string, string | number>) => string;
   607→}) {
   608→  const canvasRef = useRef<HTMLCanvasElement>(null);
   609→  const containerRef = useRef<HTMLDivElement>(null);
   610→  const imgRef = useRef<HTMLImageElement | null>(null);
   611→  
   612→  const [objects, setObjects] = useState<EditorCircle[]>(() => {
   613→    return (photo.embeddedPhotos || []).map(ep => ({
   614→      id: ep.id,
   615→      type: 'circleWithPhoto' as const,
   616→      x: ep.circleX,
   617→      y: ep.circleY,
   618→      radius: ep.circleRadius,
   619→      color: '#f97316',
   620→      imageData: ep.imageData,
   621→    }));
   622→  });
   623→  
   624→  const [currentTool, setCurrentTool] = useState<'select' | 'circle' | 'circlePhoto' | 'arrow' | 'rectangle'>('select');
   625→  const [selectedObjectId, setSelectedObjectId] = useState<string | null>(null);
   626→  const [isDragging, setIsDragging] = useState(false);
   627→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 });
   628→  const [currentColor, setCurrentColor] = useState('#f97316');
   629→  const [circleRadius, setCircleRadius] = useState(80);
   630→  const [showAddPhotoDialog, setShowAddPhotoDialog] = useState(false);
   631→  const [pendingPhotoCircle, setPendingPhotoCircle] = useState<EditorCircle | null>(null);
   632→  const [isDrawingArrow, setIsDrawingArrow] = useState(false);
   633→  const [arrowStart, setArrowStart] = useState<{ x: number; y: number } | null>(null);
   634→  const [arrowPreview, setArrowPreview] = useState<{ x: number; y: number } | null>(null);
   635→  const [isDrawingCircle, setIsDrawingCircle] = useState(false);
   636→  const [circleStart, setCircleStart] = useState<{ x: number; y: number } | null>(null);
   637→  const [circlePreview, setCirclePreview] = useState<{ x: number; y: number; radius: number } | null>(null);
   638→  const [isDrawingRect, setIsDrawingRect] = useState(false);
   639→  const [rectStart, setRectStart] = useState<{ x: number; y: number } | null>(null);
   640→  const [rectPreview, setRectPreview] = useState<{ x: number; y: number; w: number; h: number } | null>(null);
   641→  const [rotation, setRotation] = useState(0);
   642→
   643→  // Determinar qual imagem editar baseado no modo
   644→  const getActiveImageSrc = () => {
   645→    if (editMode === 'secondary') {
   646→      return photo.secondaryImageData;
   647→    }
   648→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   649→      // Criar imagem combinada lado a lado
   650→      return null; // Será tratado separadamente
   651→    }
   652→    return photo.editedImageData || photo.imageData;
   653→  };
   654→  
   655→  const imageSrc = getActiveImageSrc();
   656→  const [dualCanvasSize, setDualCanvasSize] = useState<{ width: number; height: number } | null>(null);
   657→  const [canvasSize, setCanvasSize] = useState({ width: 0, height: 0, scale: 1 });
   658→
   659→  useEffect(() => {
   660→    const container = containerRef.current;
   661→    if (!container) return;
   662→    const containerWidth = Math.min(container.clientWidth - 32, 900);
   663→    const containerHeight = 500;
   664→
   665→    // Modo duplo: carregar ambas as imagens lado a lado
   666→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   667→      const img1 = new window.Image();
   668→      const img2 = new window.Image();
   669→      
   670→      let loaded = 0;
   671→      const checkLoaded = () => {
   672→        loaded++;
   673→        if (loaded === 2) {
   674→          // Calcular dimensões para ambas as imagens lado a lado
   675→          const halfWidth = containerWidth / 2 - 10;
   676→          const scale1 = Math.min(halfWidth / img1.width, containerHeight / img1.height);
   677→          const scale2 = Math.min(halfWidth / img2.width, containerHeight / img2.height);
   678→          const scale = Math.min(scale1, scale2);
   679→          
   680→          const totalWidth = (img1.width + img2.width) * scale + 10;
   681→          const totalHeight = Math.max(img1.height, img2.height) * scale;
   682→          
   683→          setCanvasSize({ width: totalWidth, height: totalHeight, scale });
   684→          setDualCanvasSize({ width: img1.width * scale, height: img1.height * scale });
   685→          
   686→          // Criar imagem combinada temporária
   687→          const tempCanvas = document.createElement('canvas');
   688→          tempCanvas.width = img1.width + img2.width;
   689→          tempCanvas.height = Math.max(img1.height, img2.height);
   690→          const ctx = tempCanvas.getContext('2d');
   691→          if (ctx) {
   692→            ctx.drawImage(img1, 0, 0);
   693→            ctx.drawImage(img2, img1.width, 0);
   694→          }
   695→          
   696→          const combinedImg = new window.Image();
   697→          combinedImg.onload = () => {
   698→            imgRef.current = combinedImg;
   699→          };
   700→          combinedImg.src = tempCanvas.toDataURL('image/png');
   701→        }
   702→      };
   703→      
   704→      img1.onload = checkLoaded;
   705→      img2.onload = checkLoaded;
   706→      img1.src = photo.imageData;
   707→      img2.src = photo.secondaryImageData;
   708→      return;
   709→    }
   710→
   711→    // Modo normal (primário ou secundário)
   712→    if (!imageSrc) return;
   713→    const img = new window.Image();
   714→    img.onload = () => {
   715→      imgRef.current = img;
   716→      let imgWidth = img.width;
   717→      let imgHeight = img.height;
   718→      // Swap dimensions for 90/270 degree rotation
   719→      if (rotation === 90 || rotation === 270) {
   720→        [imgWidth, imgHeight] = [imgHeight, imgWidth];
   721→      }
   722→      const scale = Math.min(containerWidth / imgWidth, containerHeight / imgHeight);
   723→      setCanvasSize({ width: imgWidth * scale, height: imgHeight * scale, scale });
   724→    };
   725→    img.src = imageSrc;
   726→  }, [imageSrc, editMode, photo.imageData, photo.secondaryImageData, rotation]);
   727→
   728→  const drawCanvas = useCallback(() => {
   729→    const canvas = canvasRef.current;
   730→    const img = imgRef.current;
   731→    if (!canvas || !img || !imageSrc) return;
   732→
   733→    const ctx = canvas.getContext('2d');
   734→    if (!ctx) return;
   735→
   736→    const { width, height, scale } = canvasSize;
   737→    canvas.width = width;
   738→    canvas.height = height;
   739→    ctx.save();
   740→    if (rotation > 0) {
   741→      ctx.translate(width / 2, height / 2);
   742→      ctx.rotate((rotation * Math.PI) / 180);
   743→      ctx.translate(-width / 2, -height / 2);
   744→    }
   745→    ctx.drawImage(img, 0, 0, width, height);
   746→    ctx.restore();
   747→
   748→    objects.forEach((obj) => {
   749→      const isSelected = obj.id === selectedObjectId;
   750→      
   751→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   752→        const x = obj.x * scale;
   753→        const y = obj.y * scale;
   754→        const radius = (obj.radius || 50) * scale;
   755→
   756→        if (obj.type === 'circleWithPhoto' && obj.imageData) {
   757→          const embeddedImg = new window.Image();
   758→          embeddedImg.onload = () => {
   759→            ctx.save();
   760→            ctx.beginPath();
   761→            ctx.arc(x, y, radius - 2, 0, Math.PI * 2);
   762→            ctx.closePath();
   763→            ctx.clip();
   764→            ctx.drawImage(embeddedImg, x - radius + 2, y - radius + 2, (radius - 2) * 2, (radius - 2) * 2);
   765→            ctx.restore();
   766→            ctx.beginPath();
   767→            ctx.arc(x, y, radius, 0, Math.PI * 2);
   768→            ctx.strokeStyle = obj.color;
   769→            ctx.lineWidth = isSelected ? 4 : 3;
   770→            ctx.stroke();
   771→          };
   772→          embeddedImg.src = obj.imageData;
   773→        } else {
   774→          ctx.beginPath();
   775→          ctx.arc(x, y, radius, 0, Math.PI * 2);
   776→          ctx.strokeStyle = obj.color;
   777→          ctx.lineWidth = isSelected ? 4 : 3;
   778→          ctx.stroke();
   779→        }
   780→      }
   781→      
   782→      if (obj.type === 'arrow') {
   783→        const startX = obj.x * scale;
   784→        const startY = obj.y * scale;
   785→        const endX = (obj.endX || obj.x + 50) * scale;
   786→        const endY = (obj.endY || obj.y + 50) * scale;
   787→
   788→        ctx.beginPath();
   789→        ctx.moveTo(startX, startY);
   790→        ctx.lineTo(endX, endY);
   791→        ctx.strokeStyle = obj.color;
   792→        ctx.lineWidth = isSelected ? 4 : 3;
   793→        ctx.stroke();
   794→
   795→        const angle = Math.atan2(endY - startY, endX - startX);
   796→        ctx.beginPath();
   797→        ctx.moveTo(endX, endY);
   798→        ctx.lineTo(endX - 15 * Math.cos(angle - Math.PI / 6), endY - 15 * Math.sin(angle - Math.PI / 6));
   799→        ctx.moveTo(endX, endY);
   800→        ctx.lineTo(endX - 15 * Math.cos(angle + Math.PI / 6), endY - 15 * Math.sin(angle + Math.PI / 6));
   801→        ctx.stroke();
   802→      }
   803→    });
   804→
   805→    // Rectangle objects
   806→    objects.forEach((obj) => {
   807→      if (obj.type === 'rectangle') {
   808→        const isSelected = obj.id === selectedObjectId;
   809→        const rx = obj.x * scale;
   810→        const ry = obj.y * scale;
   811→        const rw = (obj.width || 100) * scale;
   812→        const rh = (obj.height || 80) * scale;
   813→        ctx.strokeStyle = obj.color;
   814→        ctx.lineWidth = isSelected ? 4 : 3;
   815→        ctx.strokeRect(rx, ry, rw, rh);
   816→      }
   817→    });
   818→
   819→    if (isDrawingArrow && arrowStart && arrowPreview) {
   820→      ctx.beginPath();
   821→      ctx.moveTo(arrowStart.x * scale, arrowStart.y * scale);
   822→      ctx.lineTo(arrowPreview.x * scale, arrowPreview.y * scale);
   823→      ctx.strokeStyle = currentColor;
   824→      ctx.lineWidth = 3;
   825→      ctx.setLineDash([5, 5]);
   826→      ctx.stroke();
   827→      ctx.setLineDash([]);
   828→    }
   829→
   830→    // Circle drag preview
   831→    if (isDrawingCircle && circleStart && circlePreview) {
   832→      ctx.beginPath();
   833→      ctx.arc(circlePreview.x * scale, circlePreview.y * scale, circlePreview.radius * scale, 0, Math.PI * 2);
   834→      ctx.strokeStyle = currentColor;
   835→      ctx.lineWidth = 3;
   836→      ctx.setLineDash([5, 5]);
   837→      ctx.stroke();
   838→      ctx.setLineDash([]);
   839→    }
   840→
   841→    // Rectangle drag preview
   842→    if (isDrawingRect && rectStart && rectPreview) {
   843→      ctx.strokeStyle = currentColor;
   844→      ctx.lineWidth = 3;
   845→      ctx.setLineDash([5, 5]);
   846→      ctx.strokeRect(rectPreview.x * scale, rectPreview.y * scale, rectPreview.w * scale, rectPreview.h * scale);
   847→      ctx.setLineDash([]);
   848→    }
   849→  }, [imageSrc, canvasSize, objects, selectedObjectId, isDrawingArrow, arrowStart, arrowPreview, currentColor, rotation, isDrawingCircle, circleStart, circlePreview, isDrawingRect, rectStart, rectPreview]);
   850→
   851→  useEffect(() => {
   852→    drawCanvas();
   853→  }, [drawCanvas]);
   854→
   855→  const getMousePos = (e: React.MouseEvent): { x: number; y: number } => {
   856→    const canvas = canvasRef.current;
   857→    if (!canvas) return { x: 0, y: 0 };
   858→    const rect = canvas.getBoundingClientRect();
   859→    return { x: (e.clientX - rect.left) / canvasSize.scale, y: (e.clientY - rect.top) / canvasSize.scale };
   860→  };
   861→
   862→  // Touch position helper for mobile
   863→  const getTouchPos = (e: React.TouchEvent): { x: number; y: number } => {
   864→    const canvas = canvasRef.current;
   865→    if (!canvas || e.touches.length === 0) return { x: 0, y: 0 };
   866→    const rect = canvas.getBoundingClientRect();
   867→    const touch = e.touches[0];
   868→    return { x: (touch.clientX - rect.left) / canvasSize.scale, y: (touch.clientY - rect.top) / canvasSize.scale };
   869→  };
   870→
   871→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {
   872→    for (let i = objects.length - 1; i >= 0; i--) {
   873→      const obj = objects[i];
   874→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   875→        const dist = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   876→        if (dist <= (obj.radius || 50)) return obj;
   877→      }
   878→      if (obj.type === 'arrow') {
   879→        const lineLength = Math.sqrt((obj.endX! - obj.x) ** 2 + (obj.endY! - obj.y) ** 2);
   880→        const d1 = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   881→        const d2 = Math.sqrt((x - obj.endX!) ** 2 + (y - obj.endY!) ** 2);
   882→        if (d1 + d2 <= lineLength + 15) return obj;
   883→      }
   884→      if (obj.type === 'rectangle') {
   885→        const rx = Math.min(obj.x, obj.x + (obj.width || 100));
   886→        const ry = Math.min(obj.y, obj.y + (obj.height || 80));
   887→        const rw = Math.abs(obj.width || 100);
   888→        const rh = Math.abs(obj.height || 80);
   889→        if (x >= rx - 10 && x <= rx + rw + 10 && y >= ry - 10 && y <= ry + rh + 10) return obj;
   890→      }
   891→    }
   892→    return null;
   893→  };
   894→
   895→  const handleMouseDown = (e: React.MouseEvent) => {
   896→    const pos = getMousePos(e);
   897→    if (currentTool === 'select') {
   898→      const obj = findObjectAtPosition(pos.x, pos.y);
   899→      if (obj) {
   900→        setSelectedObjectId(obj.id);
   901→        setIsDragging(true);
   902→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   903→      } else {
   904→        setSelectedObjectId(null);
   905→      }
   906→    } else if (currentTool === 'circle') {
   907→      setIsDrawingCircle(true);
   908→      setCircleStart(pos);
   909→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
   910→    } else if (currentTool === 'circlePhoto') {
   911→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
   912→      setShowAddPhotoDialog(true);
   913→    } else if (currentTool === 'arrow') {
   914→      setIsDrawingArrow(true);
   915→      setArrowStart(pos);
   916→      setArrowPreview(pos);
   917→    } else if (currentTool === 'rectangle') {
   918→      setIsDrawingRect(true);
   919→      setRectStart(pos);
   920→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
   921→    }
   922→  };
   923→
   924→  const handleMouseMove = (e: React.MouseEvent) => {
   925→    const pos = getMousePos(e);
   926→    if (isDragging && selectedObjectId) {
   927→      setObjects(prev => prev.map(obj => {
   928→        if (obj.id !== selectedObjectId) return obj;
   929→        const newX = pos.x - dragOffset.x;
   930→        const newY = pos.y - dragOffset.y;
   931→        if (obj.type === 'arrow') {
   932→          const dx = obj.endX! - obj.x;
   933→          const dy = obj.endY! - obj.y;
   934→          return { ...obj, x: newX, y: newY, endX: newX + dx, endY: newY + dy };
   935→        }
   936→        if (obj.type === 'rectangle') {
   937→          return { ...obj, x: newX, y: newY };
   938→        }
   939→        return { ...obj, x: newX, y: newY };
   940→      }));
   941→    }
   942→    if (isDrawingArrow) setArrowPreview(pos);
   943→    if (isDrawingCircle && circleStart) {
   944→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   945→      setCirclePreview({ x: circleStart.x, y: circleStart.y, radius: dist });
   946→    }
   947→    if (isDrawingRect && rectStart) {
   948→      setRectPreview({ x: rectStart.x, y: rectStart.y, w: pos.x - rectStart.x, h: pos.y - rectStart.y });
   949→    }
   950→  };
   951→
   952→  const handleMouseUp = (e: React.MouseEvent) => {
   953→    const pos = getMousePos(e);
   954→    if (isDragging) setIsDragging(false);
   955→    if (isDrawingCircle && circleStart && currentTool === 'circle') {
   956→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   957→      const finalRadius = dist > 10 ? dist : circleRadius;
   958→      const newCircle: EditorCircle = { id: `circle-${Date.now()}`, type: 'circle', x: circleStart.x, y: circleStart.y, radius: finalRadius, color: currentColor };
   959→      setObjects(prev => [...prev, newCircle]);
   960→      setSelectedObjectId(newCircle.id);
   961→      setIsDrawingCircle(false);
   962→      setCircleStart(null);
   963→      setCirclePreview(null);
   964→      setCurrentTool('select');
   965→    }
   966→    if (isDrawingRect && rectStart && currentTool === 'rectangle') {
   967→      const w = pos.x - rectStart.x;
   968→      const h = pos.y - rectStart.y;
   969→      if (Math.abs(w) > 10 && Math.abs(h) > 10) {
   970→        const newRect: EditorCircle = { id: `rect-${Date.now()}`, type: 'rectangle', x: rectStart.x, y: rectStart.y, width: w, height: h, color: currentColor };
   971→        setObjects(prev => [...prev, newRect]);
   972→        setSelectedObjectId(newRect.id);
   973→      }
   974→      setIsDrawingRect(false);
   975→      setRectStart(null);
   976→      setRectPreview(null);
   977→      setCurrentTool('select');
   978→    }
   979→    if (isDrawingArrow && arrowStart && currentTool === 'arrow') {
   980→      const distance = Math.sqrt((pos.x - arrowStart.x) ** 2 + (pos.y - arrowStart.y) ** 2);
   981→      if (distance > 10) {
   982→        const newArrow: EditorCircle = { id: `arrow-${Date.now()}`, type: 'arrow', x: arrowStart.x, y: arrowStart.y, endX: pos.x, endY: pos.y, color: currentColor };
   983→        setObjects(prev => [...prev, newArrow]);
   984→        setSelectedObjectId(newArrow.id);
   985→      }
   986→      setIsDrawingArrow(false);
   987→      setArrowStart(null);
   988→      setArrowPreview(null);
   989→      setCurrentTool('select');
   990→    }
   991→  };
   992→
   993→  // Touch event handlers for mobile
   994→  const handleTouchStart = (e: React.TouchEvent) => {
   995→    e.preventDefault();
   996→    const pos = getTouchPos(e);
   997→    if (currentTool === 'select') {
   998→      const obj = findObjectAtPosition(pos.x, pos.y);
   999→      if (obj) {
  1000→        setSelectedObjectId(obj.id);
  1001→        setIsDragging(true);
  1002→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
  1003→      } else {
  1004→        setSelectedObjectId(null);
  1005→      }
  1006→    } else if (currentTool === 'circle') {
  1007→      setIsDrawingCircle(true);
  1008→      setCircleStart(pos);

  1310→// Função para gerar imagem preta com número de série
  1311→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1312→  if (!serialNumber || serialNumber.trim() === '') return null;
  1313→  
  1314→  const canvas = document.createElement('canvas');
  1315→  canvas.width = 400;
  1316→  canvas.height = 200;
  1317→  const ctx = canvas.getContext('2d');
  1318→  
  1319→  if (!ctx) return null;
  1320→  
  1321→  // Fundo preto
  1322→  ctx.fillStyle = '#000000';
  1323→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1324→  
  1325→  // Texto branco
  1326→  ctx.fillStyle = '#FFFFFF';
  1327→  ctx.font = 'bold 32px monospace';
  1328→  ctx.textAlign = 'center';
  1329→  ctx.textBaseline = 'middle';
  1330→  
  1331→  // Desenhar o número de série
  1332→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1333→  
  1334→  return canvas.toDataURL('image/png');
  1335→}
