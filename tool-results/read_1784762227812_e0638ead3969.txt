    49→    49→

    54→    54→

   195→   195→async function getPhotoDimensions(photo: PhotoData): Promise<{ w: number; h: number; isPortrait: boolean }> {
   196→   196→  const imgData = photo.editedImageData || photo.imageData;
   197→   197→  if (!imgData) return { w: PHOTO_LANDSCAPE_W, h: PHOTO_LANDSCAPE_H, isPortrait: false };
   198→   198→  try {
   199→   199→    const info = await getImageInfo(imgData);
   200→   200→    return info.isPortrait
   201→   201→      ? { w: PHOTO_PORTRAIT_W, h: PHOTO_PORTRAIT_H, isPortrait: true }
   202→   202→      : { w: PHOTO_LANDSCAPE_W, h: PHOTO_LANDSCAPE_H, isPortrait: false };
   203→   203→  } catch {
   204→   204→    return { w: PHOTO_LANDSCAPE_W, h: PHOTO_LANDSCAPE_H, isPortrait: false };
   205→   205→  }
   206→   206→}

   208→   208→/**
   209→   209→ * Calculate x,y positions for 1 or 2 photos on a slide,
   210→   210→ * centering them horizontally.
   211→   211→ */
   212→   212→function layoutPhotos(
   213→   213→  dims1: { w: number; h: number } | null,
   214→   214→  dims2: { w: number; h: number } | null,
   215→   215→  y: number,
   216→   216→  gap: number
   217→   217→): Array<{ photo: PhotoData; x: number; y: number; w: number; h: number } | null> {
   218→   218→  const result: Array<{ photo: PhotoData; x: number; y: number; w: number; h: number } | null> = [null, null];
   219→   219→  
   220→   220→  if (!dims1 && !dims2) return result;
   221→   221→  
   222→   222→  if (dims1 && !dims2) {
   223→   223→    // Single photo: center it
   224→   224→    result[0] = { photo: null!, x: (10 - dims1.w) / 2, y, w: dims1.w, h: dims1.h };
   225→   225→    return result;
   226→   226→  }
   227→   227→  
   228→   228→  if (dims1 && dims2) {
   229→   229→    // Two photos: center them together
   230→   230→    const totalW = dims1.w + gap + dims2.w;
   231→   231→    const startX = (10 - totalW) / 2;
   232→   232→    result[0] = { photo: null!, x: startX, y, w: dims1.w, h: dims1.h };
   233→   233→    result[1] = { photo: null!, x: startX + dims1.w + gap, y, w: dims2.w, h: dims2.h };
   234→   234→    return result;
   235→   235→  }
   236→   236→  
   237→   237→  // Only dims2 (shouldn't happen)
   238→   238→  result[1] = { photo: null!, x: (10 - dims2.w) / 2, y, w: dims2.w, h: dims2.h };
   239→   239→  return result;
   240→   240→}

   242→   242→/**
   243→   243→ * Get height based on image aspect ratio for a given width.
   244→   244→ * Returns a default height if image data can't be parsed.
   245→   245→ */
   246→   246→async function getAspectRatioHeight(targetW: number, imgData: string): Promise<number> {
   247→   247→  try {
   248→   248→    const info = await getImageInfo(imgData);
   249→   249→    const ratio = info.height / info.width;
   250→   250→    const h = targetW * ratio;
   251→   251→    // Clamp between 2.5 and 5.0 inches
   252→   252→    return Math.max(2.5, Math.min(5.0, h));
   253→   253→  } catch {
   254→   254→    return PHOTO_LANDSCAPE_H;
   255→   255→  }
   256→   256→}

   258→   258→async function generateCategorySlides(
   259→   259→  pptx: pptxgen, 
   260→   260→  category: PhotoCategory, 
   261→   261→  categoryNumber: number,
   262→   262→  language: Language
   263→   263→) {
   264→   264→  const isTranslated = language !== 'pt';
   265→   265→  const photosWithImages = category.photos.filter(p => p.imageData);
   266→   266→  const slideCount = Math.ceil(photosWithImages.length / 2);
   267→   267→  
   268→   268→  // Get translated title
   269→   269→  const { title: translatedTitle } = getCategoryTranslation(category.id, language);
   270→   270→  
   271→   271→  for (let i = 0; i < slideCount; i++) {
   272→   272→    const slide = pptx.addSlide();
   273→   273→    addCategoryHeader(slide, `${categoryNumber}. ${translatedTitle.toUpperCase()}`);
   274→   274→    
   275→   275→    const photo1 = photosWithImages[i * 2];
   276→   276→    const photo2 = photosWithImages[i * 2 + 1];
   277→   277→    
   278→   278→    const isDual1 = !!(photo1?.secondaryImageData && photo1?.imageData);
   279→   279→    const isDual2 = !!(photo2?.secondaryImageData && photo2?.imageData);
   280→   280→    
   281→   281→    // Calculate widths: dual photos get wider slots
   282→   282→    let w1: number, h1: number, w2: number, h2: number;
   283→   283→    const y = 1.2;
   284→   284→    const gap = 0.2;
   285→   285→    const usableW = 10 - SLIDE_MARGIN * 2;
   286→   286→    
   287→   287→    if (isDual1 && isDual2) {
   288→   288→      // Both dual: equal wide slots
   289→   289→      w1 = (usableW - gap) / 2;
   290→   290→      w2 = w1;
   291→   291→      h1 = DUAL_PHOTO_H;
   292→   292→      h2 = DUAL_PHOTO_H;
   293→   293→    } else if (isDual1) {
   294→   294→      // Photo 1 dual, photo 2 single
   295→   295→      w1 = DUAL_SLOT_W;
   296→   296→      w2 = usableW - w1 - gap;
   297→   297→      h1 = DUAL_PHOTO_H;
   298→   298→      h2 = photo2 ? (photo2.editedImageData || photo2.imageData ? getAspectRatioHeight(w2, photo2.editedImageData || photo2.imageData!) : PHOTO_LANDSCAPE_H) : PHOTO_LANDSCAPE_H;
   299→   299→    } else if (isDual2) {
   300→   300→      // Photo 1 single, photo 2 dual
   301→   301→      w2 = DUAL_SLOT_W;
   302→   302→      w1 = usableW - w2 - gap;
   303→   303→      h1 = photo1 ? (photo1.editedImageData || photo1.imageData ? getAspectRatioHeight(w1, photo1.editedImageData || photo1.imageData!) : PHOTO_LANDSCAPE_H) : PHOTO_LANDSCAPE_H;
   304→   304→      h2 = DUAL_PHOTO_H;
   305→   305→    } else {
   306→   306→      // Both single: use orientation detection
   307→   307→      const dims1 = photo1 ? await getPhotoDimensions(photo1) : null;
   308→   308→      const dims2 = photo2 ? await getPhotoDimensions(photo2) : null;
   309→   309→      const positions = layoutPhotos(dims1, dims2, y, gap);
   310→   310→      
   311→   311→      if (photo1?.imageData && positions[0]) {
   312→   312→        await addPhotoToSlide(slide, photo1, positions[0].x, positions[0].y, positions[0].w, positions[0].h, language, String(i * 2 + 1));
   313→   313→      }
   314→   314→      if (photo2?.imageData && positions[1]) {
   315→   315→        await addPhotoToSlide(slide, photo2, positions[1].x, positions[1].y, positions[1].w, positions[1].h, language, String(i * 2 + 2));
   316→   316→      }
   317→   317→      addStandardFooter(slide);
   318→   318→      continue;
   319→   319→    }
   320→   320→    
   321→   321→    // Place photos with calculated widths
   322→   322→    const x1 = SLIDE_MARGIN;
   323→   323→    const x2 = x1 + w1 + gap;
   324→   324→    
   325→   325→    if (photo1?.imageData) {
   326→   326→      await addPhotoToSlide(slide, photo1, x1, y, w1, h1, language, String(i * 2 + 1));
   327→   327→    }
   328→   328→    if (photo2?.imageData) {
   329→   329→      await addPhotoToSlide(slide, photo2, x2, y, w2, h2, language, String(i * 2 + 2));
   330→   330→    }
   331→   331→    
   332→   332→    addStandardFooter(slide);
   333→   333→  }
   334→   334→}

   791→   791→async function generatePhotoSlides(pptx: pptxgen, photos: PhotoData[], language: Language) {
   792→   792→  const isTranslated = language !== 'pt';
   793→   793→  const photosWithImages = photos.filter(p => p.imageData);
   794→   794→  const slideCount = Math.ceil(photosWithImages.length / 2);
   795→   795→  
   796→   796→  for (let i = 0; i < slideCount; i++) {
   797→   797→    const slide = pptx.addSlide();
   798→   798→    addStandardHeader(slide, isTranslated ? 'PHOTOS' : 'FOTOS');
   799→   799→    
   800→   800→    const photo1 = photosWithImages[i * 2];
   801→   801→    const photo2 = photosWithImages[i * 2 + 1];
   802→   802→    
   803→   803→    const isDual1 = !!(photo1?.secondaryImageData && photo1?.imageData);
   804→   804→    const isDual2 = !!(photo2?.secondaryImageData && photo2?.imageData);
   805→   805→    
   806→   806→    let w1: number, h1: number, w2: number, h2: number;
   807→   807→    const y = 1.2;
   808→   808→    const gap = 0.2;
   809→   809→    const usableW = 10 - SLIDE_MARGIN * 2;
   810→   810→    
   811→   811→    if (isDual1 && isDual2) {
   812→   812→      w1 = (usableW - gap) / 2;
   813→   813→      w2 = w1;
   814→   814→      h1 = DUAL_PHOTO_H;
   815→   815→      h2 = DUAL_PHOTO_H;
   816→   816→    } else if (isDual1) {
   817→   817→      w1 = DUAL_SLOT_W;
   818→   818→      w2 = usableW - w1 - gap;
   819→   819→      h1 = DUAL_PHOTO_H;
   820→   820→      h2 = photo2 ? (photo2.editedImageData || photo2.imageData ? getAspectRatioHeight(w2, photo2.editedImageData || photo2.imageData!) : PHOTO_LANDSCAPE_H) : PHOTO_LANDSCAPE_H;
   821→   821→    } else if (isDual2) {
   822→   822→      w2 = DUAL_SLOT_W;
   823→   823→      w1 = usableW - w2 - gap;
   824→   824→      h1 = photo1 ? (photo1.editedImageData || photo1.imageData ? getAspectRatioHeight(w1, photo1.editedImageData || photo1.imageData!) : PHOTO_LANDSCAPE_H) : PHOTO_LANDSCAPE_H;
   825→   825→      h2 = DUAL_PHOTO_H;
   826→   826→    } else {
   827→   827→      const dims1 = photo1 ? await getPhotoDimensions(photo1) : null;
   828→   828→      const dims2 = photo2 ? await getPhotoDimensions(photo2) : null;
   829→   829→      const positions = layoutPhotos(dims1, dims2, y, gap);
   830→   830→      
   831→   831→      if (photo1?.imageData && positions[0]) {
   832→   832→        await addPhotoToSlide(slide, photo1, positions[0].x, positions[0].y, positions[0].w, positions[0].h, language, String(i * 2 + 1));
   833→   833→      }
   834→   834→      if (photo2?.imageData && positions[1]) {
   835→   835→        await addPhotoToSlide(slide, photo2, positions[1].x, positions[1].y, positions[1].w, positions[1].h, language, String(i * 2 + 2));
   836→   836→      }
   837→   837→      addStandardFooter(slide);
   838→   838→      continue;
   839→   839→    }
   840→   840→    
   841→   841→    const x1 = SLIDE_MARGIN;
   842→   842→    const x2 = x1 + w1 + gap;
   843→   843→    
   844→   844→    if (photo1?.imageData) {
   845→   845→      await addPhotoToSlide(slide, photo1, x1, y, w1, h1, language, String(i * 2 + 1));
   846→   846→    }
   847→   847→    if (photo2?.imageData) {
   848→   848→      await addPhotoToSlide(slide, photo2, x2, y, w2, h2, language, String(i * 2 + 2));
   849→   849→    }
   850→   850→    
   851→   851→    addStandardFooter(slide);
   852→   852→  }

  1429→  1429→function formatDateForFileName(dateString: string): string {
  1430→  1430→  if (!dateString) return new Date().toISOString().split('T')[0];
  1431→  1431→  
  1432→  1432→  // Parse date manually to avoid timezone issues
  1433→  1433→  const parts = dateString.split('-');
  1434→  1434→  if (parts.length === 3) {
  1435→  1435→    const day = parts[2].padStart(2, '0');
  1436→  1436→    const month = parts[1].padStart(2, '0');
  1437→  1437→    const year = parts[0];
  1438→  1438→    return `${day}-${month}-${year}`;
  1439→  1439→  }
  1440→  1440→  
  1441→  1441→  // Fallback: use current date
  1442→  1442→  const today = new Date();
  1443→  1443→  const day = String(today.getDate()).padStart(2, '0');
  1444→  1444→  const month = String(today.getMonth() + 1).padStart(2, '0');
  1445→  1445→  const year = today.getFullYear();
  1446→  1446→  return `${day}-${month}-${year}`;
  1447→  1447→}
