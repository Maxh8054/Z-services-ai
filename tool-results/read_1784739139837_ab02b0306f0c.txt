   868→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {

  1307→// Função para gerar imagem preta com número de série
  1308→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1309→  if (!serialNumber || serialNumber.trim() === '') return null;
  1310→  
  1311→  const canvas = document.createElement('canvas');
  1312→  canvas.width = 400;
  1313→  canvas.height = 200;
  1314→  const ctx = canvas.getContext('2d');
  1315→  
  1316→  if (!ctx) return null;
  1317→  
  1318→  // Fundo preto
  1319→  ctx.fillStyle = '#000000';
  1320→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1321→  
  1322→  // Texto branco
  1323→  ctx.fillStyle = '#FFFFFF';
  1324→  ctx.font = 'bold 32px monospace';
  1325→  ctx.textAlign = 'center';
  1326→  ctx.textBaseline = 'middle';
  1327→  
  1328→  // Desenhar o número de série
  1329→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1330→  
  1331→  return canvas.toDataURL('image/png');
  1332→}
