   765→async function addPhotoToSlide(
   766→  slide: pptxgen.Slide, 
   767→  photo: PhotoData, 
   768→  x: number, 
   769→  y: number, 
   770→  w: number, 
   771→  h: number,
   772→  language: Language,
   773→  photoRef: string
   774→) {
   775→  const isTranslated = language !== 'pt';
   776→  
   777→  // Check if we have a secondary photo (two photos side by side)
   778→  const hasSecondaryPhoto = photo.secondaryImageData && photo.imageData;
   779→  
   780→  // Calculate photo widths
   781→  const photoWidth = hasSecondaryPhoto ? (w - 0.1) / 2 : w;
   782→  
   783→  // Photo info overlay (if PN exists)
   784→  if (photo.pn) {

   796→    
   797→    slide.addText(infoText, {
   798→      x: x + 0.1,
   799→      y: y - 0.35,
   800→      w: w - 0.2,
   801→      h: 0.25,
   802→      fontSize: 9,
   803→      color: 'FFFFFF',
   804→      fontFace: 'Arial',
   805→    });
   806→  }
   807→  
   808→  // Primary Photo
   809→  if (photo.imageData) {
   810→    const processed = await maybeCoverCropImage(photo.editedImageData || photo.imageData, { targetW: photoWidth, targetH: h });
   811→    slide.addImage({
   812→      data: processed || '',
   813→      x: x,
   814→      y: y,
   815→      w: photoWidth,
   816→      h: h,
   817→    });
   818→  }
   819→  
   820→  // Secondary Photo (lado a lado)
   821→  if (photo.secondaryImageData) {
   822→    const secondX = hasSecondaryPhoto ? x + photoWidth + 0.1 : x;
   823→    const secondW = hasSecondaryPhoto ? photoWidth : w;
   824→    const processed = await maybeCoverCropImage(photo.secondaryImageData, { targetW: secondW, targetH: h });

   908→  const partRows: PartRow[] = [];

   910→  // Process each photo that should be in the table (has PN or has sub-parts)

   912→    .filter(p => photoShouldBeInPartsTable(p, additionalParts))

   915→      const photoRef = photo.pn
   916→        ? photoRefMap.get(photo.pn)

  1338→function formatDateForFileName(dateString: string): string {
  1339→  if (!dateString) return new Date().toISOString().split('T')[0];
  1340→  
  1341→  // Parse date manually to avoid timezone issues
  1342→  const parts = dateString.split('-');
  1343→  if (parts.length === 3) {
  1344→    const day = parts[2].padStart(2, '0');
  1345→    const month = parts[1].padStart(2, '0');
  1346→    const year = parts[0];
  1347→    return `${day}-${month}-${year}`;
  1348→  }
  1349→  
  1350→  // Fallback: use current date
  1351→  const today = new Date();
  1352→  const day = String(today.getDate()).padStart(2, '0');
  1353→  const month = String(today.getMonth() + 1).padStart(2, '0');
  1354→  const year = today.getFullYear();
  1355→  return `${day}-${month}-${year}`;
  1356→}

  1358→function formatDateForDisplay(dateString: string): string {
  1359→  if (!dateString) return '';
  1360→  
  1361→  // Parse date manually to avoid timezone issues (YYYY-MM-DD format)
  1362→  const parts = dateString.split('-');
  1363→  if (parts.length === 3) {
  1364→    const day = parts[2].padStart(2, '0');
  1365→    const month = parts[1].padStart(2, '0');
  1366→    const year = parts[0];
  1367→    return `${day}/${month}/${year}`;
  1368→  }
  1369→  
  1370→  return dateString;
  1371→}
