    53→function sanitizeForFilename(text: string): string {
    54→  return text
    55→    .trim()
    56→    .substring(0, 60)
    57→    .replace(/[<>:"/\\|?*\x00-\x1f]/g, '')
    58→    .replace(/\s+/g, '_')
    59→    .replace(/_+/g, '_')
    60→    .replace(/^_|_$/g, '')
    61→    || 'export';
    62→}

    64→function generateReportId(): string {
    65→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No I, O, 0, 1 to avoid confusion
    66→  let id = 'ZS-';
    67→  for (let i = 0; i < 6; i++) {
    68→    id += chars.charAt(Math.floor(Math.random() * chars.length));
    69→  }
    70→  return id;
    71→}

   102→// Componente de Menu de Ações Mobile
   103→function MobileActionMenu({
   104→  onAddPhoto,
   105→  onDataDialog,
   106→  onExportDialog,
   107→  onEmailDialog,
   108→  onClearConfirm,
   109→  reportData,
   110→  reportType,
   111→  t,
   112→}: {
   113→  onAddPhoto: () => void;
   114→  onDataDialog: () => void;
   115→  onExportDialog: () => void;
   116→  onEmailDialog: () => void;
   117→  onClearConfirm: () => void;
   118→  reportData: unknown;
   119→  reportType: 'home' | 'inspecao';
   120→  t: (key: string, params?: Record<string, string | number>) => string;
   121→}) {
   122→  const [isOpen, setIsOpen] = useState(false);
   123→  const [timeLeft, setTimeLeft] = useState(10);
   124→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   125→  const countdownRef = useRef(10);
   126→
   127→  useEffect(() => {
   128→    if (isOpen) {
   129→      // Reset countdown ref
   130→      countdownRef.current = 10;
   131→      
   132→      timerRef.current = setInterval(() => {
   133→        countdownRef.current -= 1;
   134→        const newTime = countdownRef.current;
   135→        setTimeLeft(newTime);
   136→        
   137→        if (newTime <= 0) {
   138→          setIsOpen(false);
   139→          if (timerRef.current) {
   140→            clearInterval(timerRef.current);
   141→            timerRef.current = null;
   142→          }
   143→        }
   144→      }, 1000);
   145→      
   146→      return () => {
   147→        if (timerRef.current) {
   148→          clearInterval(timerRef.current);
   149→          timerRef.current = null;
   150→        }
   151→      };
   152→    }
   153→  }, [isOpen]);
   154→
   155→  const handleAction = (action: () => void) => {
   156→    action();
   157→    setIsOpen(false);
   158→    if (timerRef.current) {
   159→      clearInterval(timerRef.current);
   160→      timerRef.current = null;
   161→    }
   162→  };
   163→
   164→  return (
   165→    <div className="md:hidden fixed bottom-4 right-4 z-50">
   166→      {/* Menu expandido */}
   167→      {isOpen && (
   168→        <div className="absolute bottom-16 right-0 bg-black/95 backdrop-blur-sm rounded-2xl p-3 shadow-2xl border border-white/10 min-w-[200px]">
   169→          <div className="text-xs text-white/60 mb-2 text-center">
   170→            Fechando em {timeLeft}s
   171→          </div>
   172→          <div className="grid grid-cols-3 gap-2">
   173→            <Button 
   174→              size="icon" 
   175→              className="rounded-full bg-green-600 hover:bg-green-700 h-12 w-12" 
   176→              onClick={() => handleAction(onAddPhoto)}
   177→              title={t('action.addPhoto')}
   178→            >
   179→              <Plus className="h-5 w-5" />
   180→            </Button>
   181→            <Button 
   182→              size="icon" 
   183→              className="rounded-full bg-blue-600 hover:bg-blue-700 h-12 w-12" 
   184→              onClick={() => handleAction(onDataDialog)}
   185→              title={t('action.importExport')}
   186→            >
   187→              <FileDown className="h-5 w-5" />
   188→            </Button>
   189→            <Button 
   190→              size="icon" 
   191→              className="rounded-full bg-purple-600 hover:bg-purple-700 h-12 w-12" 
   192→              onClick={() => handleAction(onExportDialog)}
   193→              title={t('action.download')}
   194→            >
   195→              <Download className="h-5 w-5" />
   196→            </Button>
   197→            <Button 
   198→              size="icon" 
   199→              className="rounded-full bg-orange-500 hover:bg-orange-600 h-12 w-12" 
   200→              onClick={() => handleAction(onEmailDialog)}
   201→              title="Email"
   202→            >
   203→              <Mail className="h-5 w-5" />
   204→            </Button>
   205→            <Button 
   206→              size="icon" 
   207→              className="rounded-full bg-red-600 hover:bg-red-700 h-12 w-12" 
   208→              onClick={() => handleAction(onClearConfirm)}
   209→              title={t('action.clear')}
   210→            >
   211→              <Trash2 className="h-5 w-5" />
   212→            </Button>
   213→          </div>
   214→        </div>
   215→      )}
   216→      
   217→      {/* Botão principal */}
   218→      <Button
   219→        className={`rounded-full h-14 w-14 shadow-lg ${isOpen ? 'bg-red-500 hover:bg-red-600' : 'bg-orange-500 hover:bg-orange-600'}`}
   220→        onClick={() => setIsOpen(!isOpen)}
   221→      >
   222→        {isOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   223→      </Button>
   224→    </div>
   225→  );
   226→}

   370→// Componente para a seção de peças adicionais
   371→function AdditionalPartsSection({
   372→  parentPn,
   373→  photoId,
   374→  additionalParts,
   375→  onAddPart,
   376→  onRemovePart,
   377→  onUpdatePart,
   378→  t,
   379→}: {
   380→  parentPn: string;
   381→  photoId?: string;
   382→  additionalParts: AdditionalPart[];
   383→  onAddPart: (part: AdditionalPart) => void;
   384→  onRemovePart: (id: string) => void;
   385→  onUpdatePart?: (id: string, data: Partial<AdditionalPart>) => void;
   386→  t: (key: string, params?: Record<string, string | number>) => string;
   387→}) {
   388→  const [newPn, setNewPn] = useState('');
   389→  const [newSerialNumber, setNewSerialNumber] = useState('');
   390→  const [newPartName, setNewPartName] = useState('');
   391→  const [newQuantity, setNewQuantity] = useState('');
   392→  const [newCriticality, setNewCriticality] = useState<'Alta' | 'Média' | 'Baixa' | ''>('');
   393→  const [showExcelPaste, setShowExcelPaste] = useState(false);
   394→  const [excelPasteData, setExcelPasteData] = useState('');
   395→
   396→  const photoAdditionalParts = additionalParts.filter(p =>
   397→    photoId && p.photoId === photoId
   398→  );
   399→
   400→  const handleAddPart = () => {
   401→    if (!newPn.trim() || !newPartName.trim() || !newQuantity.trim()) return;
   402→    
   403→    onAddPart({
   404→      id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   405→      pn: newPn.trim(),
   406→      serialNumber: newSerialNumber.trim(),
   407→      partName: newPartName.trim(),
   408→      quantity: newQuantity.trim(),
   409→      criticality: newCriticality,
   410→      parentPn: parentPn,
   411→      photoId: photoId || undefined,
   412→    });
   413→
   414→    setNewPn('');
   415→    setNewSerialNumber('');
   416→    setNewPartName('');
   417→    setNewQuantity('');
   418→    setNewCriticality('');
   419→  };
   420→
   421→  // Processar dados do Excel colados
   422→  const handlePasteExcel = () => {
   423→    if (!excelPasteData.trim()) return;
   424→    
   425→    // Parse dos dados - formato: Part No.      Q'ty    Part name (tab ou espaços)
   426→    const lines = excelPasteData.trim().split('\n');
   427→    
   428→    lines.forEach((line) => {
   429→      // Divide por tab ou múltiplos espaços
   430→      const parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p);
   431→      
   432→      if (parts.length >= 3) {
   433→        const pn = parts[0];
   434→        const quantity = parts[1];
   435→        const partName = parts.slice(2).join(' '); // Nome pode ter espaços
   436→        
   437→        onAddPart({
   438→          id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   439→          pn: pn,
   440→          serialNumber: '',
   441→          partName: partName,
   442→          quantity: quantity,
   443→          criticality: '',
   444→          parentPn: parentPn,
   445→          photoId: photoId || undefined,
   446→        });
   447→      }
   448→    });
   449→    
   450→    setExcelPasteData('');
   451→    setShowExcelPaste(false);
   452→  };
   453→
   454→  // Get badge color based on criticality
   455→  const getCriticalityBadge = (criticality: string) => {
   456→    switch (criticality) {
   457→      case 'Alta':
   458→        return <Badge className="bg-red-500 text-white text-xs h-5 px-2">{t('photo.criticalityHigh')}</Badge>;
   459→      case 'Média':
   460→        return <Badge className="bg-yellow-500 text-white text-xs h-5 px-2">{t('photo.criticalityMedium')}</Badge>;
   461→      case 'Baixa':
   462→        return <Badge className="bg-green-500 text-white text-xs h-5 px-2">{t('photo.criticalityLow')}</Badge>;
   463→      default:
   464→        return null;
   465→    }
   466→  };
   467→
   468→  return (
   469→    <div className="mt-3 p-3 bg-orange-50 rounded-lg border-2 border-orange-400 space-y-2">
   470→      <div className="flex items-center justify-between">
   471→        <Label className="text-sm text-orange-800 font-bold flex items-center gap-2">
   472→          <ListTree className="h-4 w-4" />
   473→          {t('subparts.partsOf', { pn: parentPn || '...' })}
   474→        </Label>
   475→        <div className="flex items-center gap-1">
   476→          {photoAdditionalParts.length > 0 && (
   477→            <Button
   478→              variant="ghost"
   479→              size="sm"
   480→              className="h-7 text-xs text-green-600 hover:text-green-800"
   481→              onClick={() => exportSubPartsToExcel(photoAdditionalParts, parentPn)}
   482→            >
   483→              <FileSpreadsheet className="h-3 w-3 mr-1" /> Baixar Excel
   484→            </Button>
   485→          )}
   486→          <Button
   487→            variant="ghost"
   488→            size="sm"
   489→            className="h-7 text-xs text-orange-600 hover:text-orange-800"
   490→            onClick={() => setShowExcelPaste(!showExcelPaste)}
   491→          >
   492→            {showExcelPaste ? 'Manual' : 'Colar Excel'}
   493→          </Button>
   494→        </div>
   495→      </div>
   496→      
   497→      {/* Excel Paste Mode */}
   498→      {showExcelPaste && (
   499→        <div className="space-y-2">
   500→          <p className="text-xs text-orange-700">
   501→            Cole os dados do Excel no formato: Part No. | Q'ty | Part name (uma linha por peça)
   502→          </p>
   503→          <Textarea
   504→            className="min-h-[80px] text-sm"
   505→            placeholder="4244836&#9;4&#9;BUCHA&#10;4244837&#9;2&#9;ANEL"
   506→            value={excelPasteData}
   507→            onChange={(e) => setExcelPasteData(e.target.value)}
   508→          />
   509→          <div className="flex gap-2">
   510→            <Button
   511→              size="sm"
   512→              className="bg-orange-500 hover:bg-orange-600"
   513→              onClick={handlePasteExcel}
   514→              disabled={!excelPasteData.trim()}
   515→            >
   516→              <Plus className="h-4 w-4 mr-1" /> Adicionar
   517→            </Button>
   518→            <Button
   519→              variant="outline"
   520→              size="sm"
   521→              onClick={() => { setExcelPasteData(''); setShowExcelPaste(false); }}
   522→            >
   523→              Cancelar
   524→            </Button>
   525→          </div>
   526→        </div>
   527→      )}
   528→      
   529→      {/* Manual Mode */}
   530→      {!showExcelPaste && (
   531→        <>
   532→      
   533→      {photoAdditionalParts.length > 0 && (
   534→        <div className="space-y-1 max-h-32 overflow-y-auto">
   535→          {photoAdditionalParts.map((part) => (
   536→            <div key={part.id} className="flex items-center gap-2 bg-white p-2 rounded border text-sm">
   537→              <span className="text-orange-600 font-mono font-bold">└─</span>
   538→              <span className="font-semibold text-gray-700">{part.pn}</span>
   539→              {part.serialNumber && (
   540→                <span className="text-gray-500">({part.serialNumber})</span>
   541→              )}
   542→              <span className="text-gray-600 flex-1 truncate">{part.partName}</span>
   543→              <Badge variant="secondary" className="text-xs h-5 px-2">{t('photo.quantity')}: {part.quantity}</Badge>
   544→              {getCriticalityBadge(part.criticality)}
   545→              <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 min-h-[44px] min-w-[44px] md:min-h-0 md:min-w-0" onClick={() => onRemovePart(part.id)}>
   546→                <X className="h-4 w-4" />
   547→              </Button>
   548→            </div>
   549→          ))}
   550→        </div>
   551→      )}
   552→
   553→      {/* Mobile layout - stacked */}
   554→      <div className="md:hidden space-y-2">
   555→        <div className="grid grid-cols-2 gap-2">
   556→          <Input className="h-11 text-base" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />

   582→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   583→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />

   585→          <SelectTrigger className="h-8 text-xs col-span-2">
   586→            <SelectValue placeholder={t('photo.criticality')} />
   587→          </SelectTrigger>
   588→          <SelectContent>
   589→            <SelectItem value="Alta">{t('photo.criticalityHigh')}</SelectItem>
   590→            <SelectItem value="Média">{t('photo.criticalityMedium')}</SelectItem>
   591→            <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>
   592→          </SelectContent>
   593→        </Select>
   594→        <Button size="icon" className="h-8 w-8 bg-orange-500 hover:bg-orange-600 col-span-2" onClick={handleAddPart} disabled={!newPn.trim() || !newPartName.trim() || !newQuantity.trim()}>
   595→          <Plus className="h-4 w-4" />
   596→        </Button>
   597→      </div>
   598→        </>
   599→      )}
   600→    </div>
   601→  );
   602→}

   607→// Editor de Foto
   608→function PhotoEditor({
   609→  photo,
   610→  editMode = 'primary',
   611→  onSave,
   612→  onClose,
   613→  t,
   614→}: {
   615→  photo: PhotoData;
   616→  editMode?: PhotoEditMode;
   617→  onSave: (data: Partial<PhotoData>) => void;
   618→  onClose: () => void;
   619→  t: (key: string, params?: Record<string, string | number>) => string;
   620→}) {
   621→  const canvasRef = useRef<HTMLCanvasElement>(null);
   622→  const containerRef = useRef<HTMLDivElement>(null);
   623→  const imgRef = useRef<HTMLImageElement | null>(null);
   624→  
   625→  const [objects, setObjects] = useState<EditorCircle[]>(() => {
   626→    return (photo.embeddedPhotos || []).map(ep => ({
   627→      id: ep.id,
   628→      type: 'circleWithPhoto' as const,
   629→      x: ep.circleX,
   630→      y: ep.circleY,
   631→      radius: ep.circleRadius,
   632→      color: '#f97316',
   633→      imageData: ep.imageData,
   634→    }));
   635→  });
   636→  
   637→  const [currentTool, setCurrentTool] = useState<'select' | 'circle' | 'circlePhoto' | 'arrow' | 'rectangle'>('select');
   638→  const [selectedObjectId, setSelectedObjectId] = useState<string | null>(null);
   639→  const [isDragging, setIsDragging] = useState(false);
   640→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 });
   641→  const [currentColor, setCurrentColor] = useState('#f97316');
   642→  const [circleRadius, setCircleRadius] = useState(80);
   643→  const [showAddPhotoDialog, setShowAddPhotoDialog] = useState(false);
   644→  const [pendingPhotoCircle, setPendingPhotoCircle] = useState<EditorCircle | null>(null);
   645→  const [isDrawingArrow, setIsDrawingArrow] = useState(false);
   646→  const [arrowStart, setArrowStart] = useState<{ x: number; y: number } | null>(null);
   647→  const [arrowPreview, setArrowPreview] = useState<{ x: number; y: number } | null>(null);
   648→  const [isDrawingCircle, setIsDrawingCircle] = useState(false);
   649→  const [circleStart, setCircleStart] = useState<{ x: number; y: number } | null>(null);
   650→  const [circlePreview, setCirclePreview] = useState<{ x: number; y: number; radius: number } | null>(null);
   651→  const [isDrawingRect, setIsDrawingRect] = useState(false);
   652→  const [rectStart, setRectStart] = useState<{ x: number; y: number } | null>(null);
   653→  const [rectPreview, setRectPreview] = useState<{ x: number; y: number; w: number; h: number } | null>(null);
   654→  const [rotation, setRotation] = useState(0);
   655→
   656→  // Determinar qual imagem editar baseado no modo
   657→  const getActiveImageSrc = () => {
   658→    if (editMode === 'secondary') {
   659→      return photo.secondaryImageData;
   660→    }
   661→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   662→      // Criar imagem combinada lado a lado
   663→      return null; // Será tratado separadamente
   664→    }
   665→    return photo.editedImageData || photo.imageData;
   666→  };
   667→  
   668→  const imageSrc = getActiveImageSrc();
   669→  const [dualCanvasSize, setDualCanvasSize] = useState<{ width: number; height: number } | null>(null);
   670→  const [canvasSize, setCanvasSize] = useState({ width: 0, height: 0, scale: 1 });
   671→
   672→  useEffect(() => {
   673→    const container = containerRef.current;
   674→    if (!container) return;
   675→    const containerWidth = Math.min(container.clientWidth - 32, 900);
   676→    const containerHeight = 500;
   677→
   678→    // Modo duplo: carregar ambas as imagens lado a lado
   679→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   680→      const img1 = new window.Image();
   681→      const img2 = new window.Image();
   682→      
   683→      let loaded = 0;
   684→      const checkLoaded = () => {
   685→        loaded++;
   686→        if (loaded === 2) {
   687→          // Calcular dimensões para ambas as imagens lado a lado
   688→          const halfWidth = containerWidth / 2 - 10;
   689→          const scale1 = Math.min(halfWidth / img1.width, containerHeight / img1.height);
   690→          const scale2 = Math.min(halfWidth / img2.width, containerHeight / img2.height);
   691→          const scale = Math.min(scale1, scale2);
   692→          
   693→          const totalWidth = (img1.width + img2.width) * scale + 10;
   694→          const totalHeight = Math.max(img1.height, img2.height) * scale;
   695→          
   696→          setCanvasSize({ width: totalWidth, height: totalHeight, scale });
   697→          setDualCanvasSize({ width: img1.width * scale, height: img1.height * scale });
   698→          
   699→          // Criar imagem combinada temporária
   700→          const tempCanvas = document.createElement('canvas');
   701→          tempCanvas.width = img1.width + img2.width;
   702→          tempCanvas.height = Math.max(img1.height, img2.height);
   703→          const ctx = tempCanvas.getContext('2d');
   704→          if (ctx) {
   705→            ctx.drawImage(img1, 0, 0);
   706→            ctx.drawImage(img2, img1.width, 0);
   707→          }
   708→          
   709→          const combinedImg = new window.Image();
   710→          combinedImg.onload = () => {
   711→            imgRef.current = combinedImg;
   712→          };
   713→          combinedImg.src = tempCanvas.toDataURL('image/png');
   714→        }
   715→      };
   716→      
   717→      img1.onload = checkLoaded;
   718→      img2.onload = checkLoaded;
   719→      img1.src = photo.imageData;
   720→      img2.src = photo.secondaryImageData;
   721→      return;
   722→    }
   723→
   724→    // Modo normal (primário ou secundário)
   725→    if (!imageSrc) return;
   726→    const img = new window.Image();
   727→    img.onload = () => {
   728→      imgRef.current = img;
   729→      let imgWidth = img.width;
   730→      let imgHeight = img.height;
   731→      // Swap dimensions for 90/270 degree rotation
   732→      if (rotation === 90 || rotation === 270) {
   733→        [imgWidth, imgHeight] = [imgHeight, imgWidth];
   734→      }
   735→      const scale = Math.min(containerWidth / imgWidth, containerHeight / imgHeight);
   736→      setCanvasSize({ width: imgWidth * scale, height: imgHeight * scale, scale });
   737→    };
   738→    img.src = imageSrc;
   739→  }, [imageSrc, editMode, photo.imageData, photo.secondaryImageData, rotation]);
   740→
   741→  const drawCanvas = useCallback(() => {
   742→    const canvas = canvasRef.current;
   743→    const img = imgRef.current;
   744→    if (!canvas || !img || !imageSrc) return;
   745→
   746→    const ctx = canvas.getContext('2d');
   747→    if (!ctx) return;
   748→
   749→    const { width, height, scale } = canvasSize;
   750→    canvas.width = width;
   751→    canvas.height = height;
   752→    ctx.save();
   753→    if (rotation > 0) {
   754→      ctx.translate(width / 2, height / 2);
   755→      ctx.rotate((rotation * Math.PI) / 180);
   756→      ctx.translate(-width / 2, -height / 2);
   757→    }
   758→    ctx.drawImage(img, 0, 0, width, height);
   759→    ctx.restore();
   760→
   761→    objects.forEach((obj) => {
   762→      const isSelected = obj.id === selectedObjectId;
   763→      
   764→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   765→        const x = obj.x * scale;
   766→        const y = obj.y * scale;
   767→        const radius = (obj.radius || 50) * scale;
   768→
   769→        if (obj.type === 'circleWithPhoto' && obj.imageData) {
   770→          const embeddedImg = new window.Image();
   771→          embeddedImg.onload = () => {
   772→            ctx.save();
   773→            ctx.beginPath();
   774→            ctx.arc(x, y, radius - 2, 0, Math.PI * 2);
   775→            ctx.closePath();
   776→            ctx.clip();
   777→            ctx.drawImage(embeddedImg, x - radius + 2, y - radius + 2, (radius - 2) * 2, (radius - 2) * 2);
   778→            ctx.restore();
   779→            ctx.beginPath();
   780→            ctx.arc(x, y, radius, 0, Math.PI * 2);
   781→            ctx.strokeStyle = obj.color;
   782→            ctx.lineWidth = isSelected ? 4 : 3;
   783→            ctx.stroke();
   784→          };
   785→          embeddedImg.src = obj.imageData;
   786→        } else {
   787→          ctx.beginPath();
   788→          ctx.arc(x, y, radius, 0, Math.PI * 2);
   789→          ctx.strokeStyle = obj.color;
   790→          ctx.lineWidth = isSelected ? 4 : 3;
   791→          ctx.stroke();
   792→        }
   793→      }
   794→      
   795→      if (obj.type === 'arrow') {
   796→        const startX = obj.x * scale;
   797→        const startY = obj.y * scale;
   798→        const endX = (obj.endX || obj.x + 50) * scale;
   799→        const endY = (obj.endY || obj.y + 50) * scale;
   800→
   801→        ctx.beginPath();
   802→        ctx.moveTo(startX, startY);
   803→        ctx.lineTo(endX, endY);
   804→        ctx.strokeStyle = obj.color;
   805→        ctx.lineWidth = isSelected ? 4 : 3;
   806→        ctx.stroke();
   807→
   808→        const angle = Math.atan2(endY - startY, endX - startX);
   809→        ctx.beginPath();
   810→        ctx.moveTo(endX, endY);
   811→        ctx.lineTo(endX - 15 * Math.cos(angle - Math.PI / 6), endY - 15 * Math.sin(angle - Math.PI / 6));
   812→        ctx.moveTo(endX, endY);
   813→        ctx.lineTo(endX - 15 * Math.cos(angle + Math.PI / 6), endY - 15 * Math.sin(angle + Math.PI / 6));
   814→        ctx.stroke();
   815→      }
   816→    });
   817→
   818→    // Rectangle objects
   819→    objects.forEach((obj) => {
   820→      if (obj.type === 'rectangle') {
   821→        const isSelected = obj.id === selectedObjectId;
   822→        const rx = obj.x * scale;
   823→        const ry = obj.y * scale;
   824→        const rw = (obj.width || 100) * scale;
   825→        const rh = (obj.height || 80) * scale;
   826→        ctx.strokeStyle = obj.color;
   827→        ctx.lineWidth = isSelected ? 4 : 3;
   828→        ctx.strokeRect(rx, ry, rw, rh);
   829→      }
   830→    });
   831→
   832→    if (isDrawingArrow && arrowStart && arrowPreview) {
   833→      ctx.beginPath();
   834→      ctx.moveTo(arrowStart.x * scale, arrowStart.y * scale);
   835→      ctx.lineTo(arrowPreview.x * scale, arrowPreview.y * scale);
   836→      ctx.strokeStyle = currentColor;
   837→      ctx.lineWidth = 3;
   838→      ctx.setLineDash([5, 5]);
   839→      ctx.stroke();
   840→      ctx.setLineDash([]);
   841→    }
   842→
   843→    // Circle drag preview
   844→    if (isDrawingCircle && circleStart && circlePreview) {
   845→      ctx.beginPath();
   846→      ctx.arc(circlePreview.x * scale, circlePreview.y * scale, circlePreview.radius * scale, 0, Math.PI * 2);
   847→      ctx.strokeStyle = currentColor;
   848→      ctx.lineWidth = 3;
   849→      ctx.setLineDash([5, 5]);
   850→      ctx.stroke();
   851→      ctx.setLineDash([]);
   852→    }
   853→
   854→    // Rectangle drag preview
   855→    if (isDrawingRect && rectStart && rectPreview) {
   856→      ctx.strokeStyle = currentColor;
   857→      ctx.lineWidth = 3;
   858→      ctx.setLineDash([5, 5]);
   859→      ctx.strokeRect(rectPreview.x * scale, rectPreview.y * scale, rectPreview.w * scale, rectPreview.h * scale);
   860→      ctx.setLineDash([]);
   861→    }
   862→  }, [imageSrc, canvasSize, objects, selectedObjectId, isDrawingArrow, arrowStart, arrowPreview, currentColor, rotation, isDrawingCircle, circleStart, circlePreview, isDrawingRect, rectStart, rectPreview]);
   863→
   864→  useEffect(() => {
   865→    drawCanvas();
   866→  }, [drawCanvas]);
   867→
   868→  const getMousePos = (e: React.MouseEvent): { x: number; y: number } => {
   869→    const canvas = canvasRef.current;
   870→    if (!canvas) return { x: 0, y: 0 };
   871→    const rect = canvas.getBoundingClientRect();
   872→    return { x: (e.clientX - rect.left) / canvasSize.scale, y: (e.clientY - rect.top) / canvasSize.scale };
   873→  };
   874→
   875→  // Touch position helper for mobile
   876→  const getTouchPos = (e: React.TouchEvent): { x: number; y: number } => {
   877→    const canvas = canvasRef.current;
   878→    if (!canvas || e.touches.length === 0) return { x: 0, y: 0 };
   879→    const rect = canvas.getBoundingClientRect();
   880→    const touch = e.touches[0];
   881→    return { x: (touch.clientX - rect.left) / canvasSize.scale, y: (touch.clientY - rect.top) / canvasSize.scale };
   882→  };
   883→
   884→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {
   885→    for (let i = objects.length - 1; i >= 0; i--) {
   886→      const obj = objects[i];
   887→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   888→        const dist = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   889→        if (dist <= (obj.radius || 50)) return obj;
   890→      }
   891→      if (obj.type === 'arrow') {
   892→        const lineLength = Math.sqrt((obj.endX! - obj.x) ** 2 + (obj.endY! - obj.y) ** 2);
   893→        const d1 = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   894→        const d2 = Math.sqrt((x - obj.endX!) ** 2 + (y - obj.endY!) ** 2);
   895→        if (d1 + d2 <= lineLength + 15) return obj;
   896→      }
   897→      if (obj.type === 'rectangle') {
   898→        const rx = Math.min(obj.x, obj.x + (obj.width || 100));
   899→        const ry = Math.min(obj.y, obj.y + (obj.height || 80));
   900→        const rw = Math.abs(obj.width || 100);
   901→        const rh = Math.abs(obj.height || 80);
   902→        if (x >= rx - 10 && x <= rx + rw + 10 && y >= ry - 10 && y <= ry + rh + 10) return obj;
   903→      }
   904→    }
   905→    return null;
   906→  };
   907→
   908→  const handleMouseDown = (e: React.MouseEvent) => {
   909→    const pos = getMousePos(e);
   910→    if (currentTool === 'select') {
   911→      const obj = findObjectAtPosition(pos.x, pos.y);
   912→      if (obj) {
   913→        setSelectedObjectId(obj.id);
   914→        setIsDragging(true);
   915→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   916→      } else {
   917→        setSelectedObjectId(null);
   918→      }
   919→    } else if (currentTool === 'circle') {
   920→      setIsDrawingCircle(true);
   921→      setCircleStart(pos);
   922→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
   923→    } else if (currentTool === 'circlePhoto') {
   924→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
   925→      setShowAddPhotoDialog(true);
   926→    } else if (currentTool === 'arrow') {
   927→      setIsDrawingArrow(true);
   928→      setArrowStart(pos);
   929→      setArrowPreview(pos);
   930→    } else if (currentTool === 'rectangle') {
   931→      setIsDrawingRect(true);
   932→      setRectStart(pos);
   933→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
   934→    }
   935→  };
   936→
   937→  const handleMouseMove = (e: React.MouseEvent) => {
   938→    const pos = getMousePos(e);
   939→    if (isDragging && selectedObjectId) {
   940→      setObjects(prev => prev.map(obj => {
   941→        if (obj.id !== selectedObjectId) return obj;
   942→        const newX = pos.x - dragOffset.x;
   943→        const newY = pos.y - dragOffset.y;
   944→        if (obj.type === 'arrow') {
   945→          const dx = obj.endX! - obj.x;
   946→          const dy = obj.endY! - obj.y;
   947→          return { ...obj, x: newX, y: newY, endX: newX + dx, endY: newY + dy };
   948→        }
   949→        if (obj.type === 'rectangle') {
   950→          return { ...obj, x: newX, y: newY };
   951→        }
   952→        return { ...obj, x: newX, y: newY };
   953→      }));
   954→    }
   955→    if (isDrawingArrow) setArrowPreview(pos);
   956→    if (isDrawingCircle && circleStart) {
   957→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   958→      setCirclePreview({ x: circleStart.x, y: circleStart.y, radius: dist });
   959→    }
   960→    if (isDrawingRect && rectStart) {
   961→      setRectPreview({ x: rectStart.x, y: rectStart.y, w: pos.x - rectStart.x, h: pos.y - rectStart.y });
   962→    }
   963→  };
   964→
   965→  const handleMouseUp = (e: React.MouseEvent) => {
   966→    const pos = getMousePos(e);
   967→    if (isDragging) setIsDragging(false);
   968→    if (isDrawingCircle && circleStart && currentTool === 'circle') {
   969→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   970→      const finalRadius = dist > 10 ? dist : circleRadius;
   971→      const newCircle: EditorCircle = { id: `circle-${Date.now()}`, type: 'circle', x: circleStart.x, y: circleStart.y, radius: finalRadius, color: currentColor };
   972→      setObjects(prev => [...prev, newCircle]);
   973→      setSelectedObjectId(newCircle.id);
   974→      setIsDrawingCircle(false);
   975→      setCircleStart(null);
   976→      setCirclePreview(null);
   977→      setCurrentTool('select');
   978→    }
   979→    if (isDrawingRect && rectStart && currentTool === 'rectangle') {
   980→      const w = pos.x - rectStart.x;
   981→      const h = pos.y - rectStart.y;
   982→      if (Math.abs(w) > 10 && Math.abs(h) > 10) {
   983→        const newRect: EditorCircle = { id: `rect-${Date.now()}`, type: 'rectangle', x: rectStart.x, y: rectStart.y, width: w, height: h, color: currentColor };
   984→        setObjects(prev => [...prev, newRect]);
   985→        setSelectedObjectId(newRect.id);
   986→      }
   987→      setIsDrawingRect(false);
   988→      setRectStart(null);
   989→      setRectPreview(null);
   990→      setCurrentTool('select');
   991→    }
   992→    if (isDrawingArrow && arrowStart && currentTool === 'arrow') {
   993→      const distance = Math.sqrt((pos.x - arrowStart.x) ** 2 + (pos.y - arrowStart.y) ** 2);
   994→      if (distance > 10) {
   995→        const newArrow: EditorCircle = { id: `arrow-${Date.now()}`, type: 'arrow', x: arrowStart.x, y: arrowStart.y, endX: pos.x, endY: pos.y, color: currentColor };
   996→        setObjects(prev => [...prev, newArrow]);
   997→        setSelectedObjectId(newArrow.id);
   998→      }
   999→      setIsDrawingArrow(false);
  1000→      setArrowStart(null);
  1001→      setArrowPreview(null);
  1002→      setCurrentTool('select');
  1003→    }
  1004→  };
  1005→
  1006→  // Touch event handlers for mobile
  1007→  const handleTouchStart = (e: React.TouchEvent) => {
  1008→    e.preventDefault();
  1009→    const pos = getTouchPos(e);
  1010→    if (currentTool === 'select') {
  1011→      const obj = findObjectAtPosition(pos.x, pos.y);
  1012→      if (obj) {
  1013→        setSelectedObjectId(obj.id);
  1014→        setIsDragging(true);

  1327→// Função para gerar imagem preta com número de série
  1328→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1329→  if (!serialNumber || serialNumber.trim() === '') return null;
  1330→  
  1331→  const canvas = document.createElement('canvas');
  1332→  canvas.width = 400;
  1333→  canvas.height = 200;
  1334→  const ctx = canvas.getContext('2d');
  1335→  
  1336→  if (!ctx) return null;
  1337→  
  1338→  // Fundo preto
  1339→  ctx.fillStyle = '#000000';
  1340→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1341→  
  1342→  // Texto branco
  1343→  ctx.fillStyle = '#FFFFFF';
  1344→  ctx.font = 'bold 32px monospace';
  1345→  ctx.textAlign = 'center';
  1346→  ctx.textBaseline = 'middle';
  1347→  
  1348→  // Desenhar o número de série
  1349→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1350→  
  1351→  return canvas.toDataURL('image/png');
  1352→}
