    49→function sanitizeForFilename(text: string): string {
    50→  return text
    51→    .trim()
    52→    .substring(0, 60)
    53→    .replace(/[<>:"/\\|?*\x00-\x1f]/g, '')
    54→    .replace(/\s+/g, '_')
    55→    .replace(/_+/g, '_')
    56→    .replace(/^_|_$/g, '')
    57→    || 'export';
    58→}

    60→function generateReportId(): string {
    61→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No I, O, 0, 1 to avoid confusion
    62→  let id = 'ZS-';
    63→  for (let i = 0; i < 6; i++) {
    64→    id += chars.charAt(Math.floor(Math.random() * chars.length));
    65→  }
    66→  return id;
    67→}

    98→// Componente de Menu de Ações Mobile
    99→function MobileActionMenu({
   100→  onAddPhoto,
   101→  onDataDialog,
   102→  onExportDialog,
   103→  onEmailDialog,
   104→  onClearConfirm,
   105→  reportData,
   106→  reportType,
   107→  t,
   108→}: {
   109→  onAddPhoto: () => void;
   110→  onDataDialog: () => void;
   111→  onExportDialog: () => void;
   112→  onEmailDialog: () => void;
   113→  onClearConfirm: () => void;
   114→  reportData: unknown;
   115→  reportType: 'home' | 'inspecao';
   116→  t: (key: string, params?: Record<string, string | number>) => string;
   117→}) {
   118→  const [isOpen, setIsOpen] = useState(false);
   119→  const [timeLeft, setTimeLeft] = useState(10);
   120→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   121→  const countdownRef = useRef(10);
   122→
   123→  useEffect(() => {
   124→    if (isOpen) {
   125→      // Reset countdown ref
   126→      countdownRef.current = 10;
   127→      
   128→      timerRef.current = setInterval(() => {
   129→        countdownRef.current -= 1;
   130→        const newTime = countdownRef.current;
   131→        setTimeLeft(newTime);
   132→        
   133→        if (newTime <= 0) {
   134→          setIsOpen(false);
   135→          if (timerRef.current) {
   136→            clearInterval(timerRef.current);
   137→            timerRef.current = null;
   138→          }
   139→        }
   140→      }, 1000);
   141→      
   142→      return () => {
   143→        if (timerRef.current) {
   144→          clearInterval(timerRef.current);
   145→          timerRef.current = null;
   146→        }
   147→      };
   148→    }
   149→  }, [isOpen]);
   150→
   151→  const handleAction = (action: () => void) => {
   152→    action();
   153→    setIsOpen(false);
   154→    if (timerRef.current) {
   155→      clearInterval(timerRef.current);
   156→      timerRef.current = null;
   157→    }
   158→  };
   159→
   160→  return (
   161→    <div className="md:hidden fixed bottom-4 right-4 z-50">
   162→      {/* Menu expandido */}
   163→      {isOpen && (
   164→        <div className="absolute bottom-16 right-0 bg-black/95 backdrop-blur-sm rounded-2xl p-3 shadow-2xl border border-white/10 min-w-[200px]">
   165→          <div className="text-xs text-white/60 mb-2 text-center">
   166→            Fechando em {timeLeft}s
   167→          </div>
   168→          <div className="grid grid-cols-3 gap-2">
   169→            <Button 
   170→              size="icon" 
   171→              className="rounded-full bg-green-600 hover:bg-green-700 h-12 w-12" 
   172→              onClick={() => handleAction(onAddPhoto)}
   173→              title={t('action.addPhoto')}
   174→            >
   175→              <Plus className="h-5 w-5" />
   176→            </Button>
   177→            <Button 
   178→              size="icon" 
   179→              className="rounded-full bg-blue-600 hover:bg-blue-700 h-12 w-12" 
   180→              onClick={() => handleAction(onDataDialog)}
   181→              title={t('action.importExport')}
   182→            >
   183→              <FileDown className="h-5 w-5" />
   184→            </Button>
   185→            <Button 
   186→              size="icon" 
   187→              className="rounded-full bg-purple-600 hover:bg-purple-700 h-12 w-12" 
   188→              onClick={() => handleAction(onExportDialog)}
   189→              title={t('action.download')}
   190→            >
   191→              <Download className="h-5 w-5" />
   192→            </Button>
   193→            <Button 
   194→              size="icon" 
   195→              className="rounded-full bg-orange-500 hover:bg-orange-600 h-12 w-12" 
   196→              onClick={() => handleAction(onEmailDialog)}
   197→              title="Email"
   198→            >
   199→              <Mail className="h-5 w-5" />
   200→            </Button>
   201→            <Button 
   202→              size="icon" 
   203→              className="rounded-full bg-red-600 hover:bg-red-700 h-12 w-12" 
   204→              onClick={() => handleAction(onClearConfirm)}
   205→              title={t('action.clear')}
   206→            >
   207→              <Trash2 className="h-5 w-5" />
   208→            </Button>
   209→          </div>
   210→        </div>
   211→      )}
   212→      
   213→      {/* Botão principal */}
   214→      <Button
   215→        className={`rounded-full h-14 w-14 shadow-lg ${isOpen ? 'bg-red-500 hover:bg-red-600' : 'bg-orange-500 hover:bg-orange-600'}`}
   216→        onClick={() => setIsOpen(!isOpen)}
   217→      >
   218→        {isOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   219→      </Button>
   220→    </div>
   221→  );
   222→}

   366→// Componente para a seção de peças adicionais
   367→function AdditionalPartsSection({
   368→  parentPn,
   369→  additionalParts,
   370→  onAddPart,
   371→  onRemovePart,
   372→  onUpdatePart,
   373→  t,
   374→}: {
   375→  parentPn: string;
   376→  additionalParts: AdditionalPart[];
   377→  onAddPart: (part: AdditionalPart) => void;
   378→  onRemovePart: (id: string) => void;
   379→  onUpdatePart?: (id: string, data: Partial<AdditionalPart>) => void;
   380→  t: (key: string, params?: Record<string, string | number>) => string;
   381→}) {
   382→  const [newPn, setNewPn] = useState('');
   383→  const [newSerialNumber, setNewSerialNumber] = useState('');
   384→  const [newPartName, setNewPartName] = useState('');
   385→  const [newQuantity, setNewQuantity] = useState('');
   386→  const [newCriticality, setNewCriticality] = useState<'Alta' | 'Média' | 'Baixa' | ''>('');
   387→  const [showExcelPaste, setShowExcelPaste] = useState(false);
   388→  const [excelPasteData, setExcelPasteData] = useState('');
   389→
   390→  const photoAdditionalParts = additionalParts.filter(p => p.parentPn === parentPn);
   391→
   392→  const handleAddPart = () => {
   393→    if (!newPn.trim() || !newPartName.trim() || !newQuantity.trim()) return;
   394→    
   395→    onAddPart({
   396→      id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   397→      pn: newPn.trim(),
   398→      serialNumber: newSerialNumber.trim(),
   399→      partName: newPartName.trim(),
   400→      quantity: newQuantity.trim(),
   401→      criticality: newCriticality,
   402→      parentPn: parentPn,
   403→    });
   404→
   405→    setNewPn('');
   406→    setNewSerialNumber('');
   407→    setNewPartName('');
   408→    setNewQuantity('');
   409→    setNewCriticality('');
   410→  };
   411→
   412→  // Processar dados do Excel colados
   413→  const handlePasteExcel = () => {
   414→    if (!excelPasteData.trim()) return;
   415→    
   416→    // Parse dos dados - formato: Part No.      Q'ty    Part name (tab ou espaços)
   417→    const lines = excelPasteData.trim().split('\n');
   418→    
   419→    lines.forEach((line) => {
   420→      // Divide por tab ou múltiplos espaços
   421→      const parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p);
   422→      
   423→      if (parts.length >= 3) {
   424→        const pn = parts[0];
   425→        const quantity = parts[1];
   426→        const partName = parts.slice(2).join(' '); // Nome pode ter espaços
   427→        
   428→        onAddPart({
   429→          id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   430→          pn: pn,
   431→          serialNumber: '',
   432→          partName: partName,
   433→          quantity: quantity,
   434→          criticality: '',
   435→          parentPn: parentPn,
   436→        });
   437→      }
   438→    });
   439→    
   440→    setExcelPasteData('');
   441→    setShowExcelPaste(false);
   442→  };
   443→
   444→  // Get badge color based on criticality
   445→  const getCriticalityBadge = (criticality: string) => {
   446→    switch (criticality) {
   447→      case 'Alta':
   448→        return <Badge className="bg-red-500 text-white text-xs h-5 px-2">{t('photo.criticalityHigh')}</Badge>;
   449→      case 'Média':
   450→        return <Badge className="bg-yellow-500 text-white text-xs h-5 px-2">{t('photo.criticalityMedium')}</Badge>;
   451→      case 'Baixa':
   452→        return <Badge className="bg-green-500 text-white text-xs h-5 px-2">{t('photo.criticalityLow')}</Badge>;
   453→      default:
   454→        return null;
   455→    }
   456→  };
   457→
   458→  return (
   459→    <div className="mt-3 p-3 bg-orange-50 rounded-lg border-2 border-orange-400 space-y-2">
   460→      <div className="flex items-center justify-between">
   461→        <Label className="text-sm text-orange-800 font-bold flex items-center gap-2">
   462→          <ListTree className="h-4 w-4" />
   463→          {t('subparts.partsOf', { pn: parentPn || '...' })}
   464→        </Label>
   465→        <Button
   466→          variant="ghost"
   467→          size="sm"
   468→          className="h-7 text-xs text-orange-600 hover:text-orange-800"
   469→          onClick={() => setShowExcelPaste(!showExcelPaste)}
   470→        >
   471→          {showExcelPaste ? 'Manual' : 'Colar Excel'}
   472→        </Button>
   473→      </div>
   474→      
   475→      {/* Excel Paste Mode */}
   476→      {showExcelPaste && (
   477→        <div className="space-y-2">
   478→          <p className="text-xs text-orange-700">
   479→            Cole os dados do Excel no formato: Part No. | Q'ty | Part name (uma linha por peça)
   480→          </p>
   481→          <Textarea
   482→            className="min-h-[80px] text-sm"
   483→            placeholder="4244836&#9;4&#9;BUCHA&#10;4244837&#9;2&#9;ANEL"
   484→            value={excelPasteData}
   485→            onChange={(e) => setExcelPasteData(e.target.value)}
   486→          />
   487→          <div className="flex gap-2">
   488→            <Button
   489→              size="sm"
   490→              className="bg-orange-500 hover:bg-orange-600"
   491→              onClick={handlePasteExcel}
   492→              disabled={!excelPasteData.trim()}
   493→            >
   494→              <Plus className="h-4 w-4 mr-1" /> Adicionar
   495→            </Button>
   496→            <Button
   497→              variant="outline"
   498→              size="sm"
   499→              onClick={() => { setExcelPasteData(''); setShowExcelPaste(false); }}
   500→            >
   501→              Cancelar
   502→            </Button>
   503→          </div>
   504→        </div>
   505→      )}
   506→      
   507→      {/* Manual Mode */}
   508→      {!showExcelPaste && (
   509→        <>
   510→      
   511→      {photoAdditionalParts.length > 0 && (
   512→        <div className="space-y-1 max-h-32 overflow-y-auto">
   513→          {photoAdditionalParts.map((part) => (
   514→            <div key={part.id} className="flex items-center gap-2 bg-white p-2 rounded border text-sm">
   515→              <span className="text-orange-600 font-mono font-bold">└─</span>
   516→              <span className="font-semibold text-gray-700">{part.pn}</span>
   517→              {part.serialNumber && (
   518→                <span className="text-gray-500">({part.serialNumber})</span>
   519→              )}
   520→              <span className="text-gray-600 flex-1 truncate">{part.partName}</span>
   521→              <Badge variant="secondary" className="text-xs h-5 px-2">{t('photo.quantity')}: {part.quantity}</Badge>
   522→              {getCriticalityBadge(part.criticality)}
   523→              <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 min-h-[44px] min-w-[44px] md:min-h-0 md:min-w-0" onClick={() => onRemovePart(part.id)}>
   524→                <X className="h-4 w-4" />
   525→              </Button>
   526→            </div>
   527→          ))}
   528→        </div>
   529→      )}
   530→
   531→      {/* Mobile layout - stacked */}
   532→      <div className="md:hidden space-y-2">
   533→        <div className="grid grid-cols-2 gap-2">
   534→          <Input className="h-11 text-base" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />
   535→          <Input className="h-11 text-base" placeholder={t('photo.serialPlaceholder')} value={newSerialNumber} onChange={(e) => setNewSerialNumber(e.target.value)} />
   536→        </div>
   537→        <Input className="h-11 text-base" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   538→        <div className="grid grid-cols-2 gap-2">
   539→          <Input className="h-11 text-base" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />
   540→          <Select value={newCriticality} onValueChange={(v) => setNewCriticality(v as 'Alta' | 'Média' | 'Baixa' | '')}>
   541→            <SelectTrigger className="h-11 text-base">
   542→              <SelectValue placeholder={t('photo.criticality')} />
   543→            </SelectTrigger>
   544→            <SelectContent>
   545→              <SelectItem value="Alta">{t('photo.criticalityHigh')}</SelectItem>
   546→              <SelectItem value="Média">{t('photo.criticalityMedium')}</SelectItem>
   547→              <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>
   548→            </SelectContent>
   549→          </Select>
   550→        </div>

   555→
   556→      {/* Desktop layout - inline grid */}
   557→      <div className="hidden md:grid grid-cols-12 gap-2">
   558→        <Input className="h-8 text-sm col-span-2" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />
   559→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.serialPlaceholder')} value={newSerialNumber} onChange={(e) => setNewSerialNumber(e.target.value)} />
   560→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   561→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />
   562→        <Select value={newCriticality} onValueChange={(v) => setNewCriticality(v as 'Alta' | 'Média' | 'Baixa' | '')}>
   563→          <SelectTrigger className="h-8 text-xs col-span-2">
   564→            <SelectValue placeholder={t('photo.criticality')} />
   565→          </SelectTrigger>
   566→          <SelectContent>
   567→            <SelectItem value="Alta">{t('photo.criticalityHigh')}</SelectItem>
   568→            <SelectItem value="Média">{t('photo.criticalityMedium')}</SelectItem>
   569→            <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>
   570→          </SelectContent>
   571→        </Select>
   572→        <Button size="icon" className="h-8 w-8 bg-orange-500 hover:bg-orange-600 col-span-2" onClick={handleAddPart} disabled={!newPn.trim() || !newPartName.trim() || !newQuantity.trim()}>
   573→          <Plus className="h-4 w-4" />
   574→        </Button>
   575→      </div>
   576→        </>
   577→      )}
   578→    </div>
   579→  );
   580→}

   625→  const [arrowPreview, setArrowPreview] = useState<{ x: number; y: number } | null>(null);
   626→  const [isDrawingCircle, setIsDrawingCircle] = useState(false);
   627→  const [circleStart, setCircleStart] = useState<{ x: number; y: number } | null>(null);

   640→      // Criar imagem combinada lado a lado
   641→      return null; // Será tratado separadamente
   642→    }
   643→    return photo.editedImageData || photo.imageData;
   644→  };

   646→  const imageSrc = getActiveImageSrc();
   647→  const [dualCanvasSize, setDualCanvasSize] = useState<{ width: number; height: number } | null>(null);
   648→  const [canvasSize, setCanvasSize] = useState({ width: 0, height: 0, scale: 1 });
   649→
   650→  useEffect(() => {
   651→    const container = containerRef.current;
   652→    if (!container) return;
   653→    const containerWidth = Math.min(container.clientWidth - 32, 900);
   654→    const containerHeight = 500;
   655→
   656→    // Modo duplo: carregar ambas as imagens lado a lado
   657→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   658→      const img1 = new window.Image();
   659→      const img2 = new window.Image();
   660→      
   661→      let loaded = 0;
   662→      const checkLoaded = () => {
   663→        loaded++;
   664→        if (loaded === 2) {
   665→          // Calcular dimensões para ambas as imagens lado a lado
   666→          const halfWidth = containerWidth / 2 - 10;
   667→          const scale1 = Math.min(halfWidth / img1.width, containerHeight / img1.height);

   669→          const scale = Math.min(scale1, scale2);
   670→          
   671→          const totalWidth = (img1.width + img2.width) * scale + 10;
   672→          const totalHeight = Math.max(img1.height, img2.height) * scale;
   673→          
   674→          setCanvasSize({ width: totalWidth, height: totalHeight, scale });
   675→          setDualCanvasSize({ width: img1.width * scale, height: img1.height * scale });
   676→          
   677→          // Criar imagem combinada temporária
   678→          const tempCanvas = document.createElement('canvas');
   679→          tempCanvas.width = img1.width + img2.width;
   680→          tempCanvas.height = Math.max(img1.height, img2.height);
   681→          const ctx = tempCanvas.getContext('2d');
   682→          if (ctx) {
   683→            ctx.drawImage(img1, 0, 0);
   684→            ctx.drawImage(img2, img1.width, 0);
   685→          }
   686→          
   687→          const combinedImg = new window.Image();
   688→          combinedImg.onload = () => {
   689→            imgRef.current = combinedImg;
   690→          };
   691→          combinedImg.src = tempCanvas.toDataURL('image/png');
   692→        }
   693→      };
   694→      
   695→      img1.onload = checkLoaded;
   696→      img2.onload = checkLoaded;
   697→      img1.src = photo.imageData;
   698→      img2.src = photo.secondaryImageData;
   699→      return;
   700→    }
   701→
   702→    // Modo normal (primário ou secundário)
   703→    if (!imageSrc) return;
   704→    const img = new window.Image();
   705→    img.onload = () => {
   706→      imgRef.current = img;
   707→      let imgWidth = img.width;
   708→      let imgHeight = img.height;
   709→      // Swap dimensions for 90/270 degree rotation
   710→      if (rotation === 90 || rotation === 270) {
   711→        [imgWidth, imgHeight] = [imgHeight, imgWidth];
   712→      }
   713→      const scale = Math.min(containerWidth / imgWidth, containerHeight / imgHeight);
   714→      setCanvasSize({ width: imgWidth * scale, height: imgHeight * scale, scale });
   715→    };
   716→    img.src = imageSrc;
   717→  }, [imageSrc, editMode, photo.imageData, photo.secondaryImageData, rotation]);
   718→
   719→  const drawCanvas = useCallback(() => {
   720→    const canvas = canvasRef.current;
   721→    const img = imgRef.current;
   722→    if (!canvas || !img || !imageSrc) return;
   723→
   724→    const ctx = canvas.getContext('2d');
   725→    if (!ctx) return;
   726→
   727→    const { width, height, scale } = canvasSize;
   728→    canvas.width = width;
   729→    canvas.height = height;
   730→    ctx.save();
   731→    if (rotation > 0) {
   732→      ctx.translate(width / 2, height / 2);
   733→      ctx.rotate((rotation * Math.PI) / 180);
   734→      ctx.translate(-width / 2, -height / 2);
   735→    }
   736→    ctx.drawImage(img, 0, 0, width, height);
   737→    ctx.restore();
   738→
   739→    objects.forEach((obj) => {
   740→      const isSelected = obj.id === selectedObjectId;
   741→      
   742→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   743→        const x = obj.x * scale;
   744→        const y = obj.y * scale;
   745→        const radius = (obj.radius || 50) * scale;
   746→
   747→        if (obj.type === 'circleWithPhoto' && obj.imageData) {
   748→          const embeddedImg = new window.Image();
   749→          embeddedImg.onload = () => {
   750→            ctx.save();
   751→            ctx.beginPath();
   752→            ctx.arc(x, y, radius - 2, 0, Math.PI * 2);
   753→            ctx.closePath();
   754→            ctx.clip();
   755→            ctx.drawImage(embeddedImg, x - radius + 2, y - radius + 2, (radius - 2) * 2, (radius - 2) * 2);
   756→            ctx.restore();
   757→            ctx.beginPath();
   758→            ctx.arc(x, y, radius, 0, Math.PI * 2);
   759→            ctx.strokeStyle = obj.color;
   760→            ctx.lineWidth = isSelected ? 4 : 3;
   761→            ctx.stroke();
   762→          };
   763→          embeddedImg.src = obj.imageData;
   764→        } else {
   765→          ctx.beginPath();
   766→          ctx.arc(x, y, radius, 0, Math.PI * 2);
   767→          ctx.strokeStyle = obj.color;
   768→          ctx.lineWidth = isSelected ? 4 : 3;
   769→          ctx.stroke();
   770→        }
   771→      }
   772→      
   773→      if (obj.type === 'arrow') {
   774→        const startX = obj.x * scale;
   775→        const startY = obj.y * scale;
   776→        const endX = (obj.endX || obj.x + 50) * scale;
   777→        const endY = (obj.endY || obj.y + 50) * scale;
   778→
   779→        ctx.beginPath();
   780→        ctx.moveTo(startX, startY);
   781→        ctx.lineTo(endX, endY);
   782→        ctx.strokeStyle = obj.color;
   783→        ctx.lineWidth = isSelected ? 4 : 3;
   784→        ctx.stroke();
   785→
   786→        const angle = Math.atan2(endY - startY, endX - startX);
   787→        ctx.beginPath();
   788→        ctx.moveTo(endX, endY);
   789→        ctx.lineTo(endX - 15 * Math.cos(angle - Math.PI / 6), endY - 15 * Math.sin(angle - Math.PI / 6));
   790→        ctx.moveTo(endX, endY);
   791→        ctx.lineTo(endX - 15 * Math.cos(angle + Math.PI / 6), endY - 15 * Math.sin(angle + Math.PI / 6));
   792→        ctx.stroke();
   793→      }
   794→    });
   795→
   796→    // Rectangle objects
   797→    objects.forEach((obj) => {
   798→      if (obj.type === 'rectangle') {
   799→        const isSelected = obj.id === selectedObjectId;
   800→        const rx = obj.x * scale;
   801→        const ry = obj.y * scale;
   802→        const rw = (obj.width || 100) * scale;
   803→        const rh = (obj.height || 80) * scale;
   804→        ctx.strokeStyle = obj.color;
   805→        ctx.lineWidth = isSelected ? 4 : 3;
   806→        ctx.strokeRect(rx, ry, rw, rh);
   807→      }
   808→    });
   809→
   810→    if (isDrawingArrow && arrowStart && arrowPreview) {
   811→      ctx.beginPath();
   812→      ctx.moveTo(arrowStart.x * scale, arrowStart.y * scale);
   813→      ctx.lineTo(arrowPreview.x * scale, arrowPreview.y * scale);
   814→      ctx.strokeStyle = currentColor;
   815→      ctx.lineWidth = 3;
   816→      ctx.setLineDash([5, 5]);
   817→      ctx.stroke();
   818→      ctx.setLineDash([]);
   819→    }
   820→
   821→    // Circle drag preview
   822→    if (isDrawingCircle && circleStart && circlePreview) {
   823→      ctx.beginPath();
   824→      ctx.arc(circlePreview.x * scale, circlePreview.y * scale, circlePreview.radius * scale, 0, Math.PI * 2);
   825→      ctx.strokeStyle = currentColor;
   826→      ctx.lineWidth = 3;
   827→      ctx.setLineDash([5, 5]);
   828→      ctx.stroke();
   829→      ctx.setLineDash([]);
   830→    }
   831→
   832→    // Rectangle drag preview
   833→    if (isDrawingRect && rectStart && rectPreview) {
   834→      ctx.strokeStyle = currentColor;
   835→      ctx.lineWidth = 3;
   836→      ctx.setLineDash([5, 5]);
   837→      ctx.strokeRect(rectPreview.x * scale, rectPreview.y * scale, rectPreview.w * scale, rectPreview.h * scale);
   838→      ctx.setLineDash([]);
   839→    }
   840→  }, [imageSrc, canvasSize, objects, selectedObjectId, isDrawingArrow, arrowStart, arrowPreview, currentColor, rotation, isDrawingCircle, circleStart, circlePreview, isDrawingRect, rectStart, rectPreview]);
   841→
   842→  useEffect(() => {
   843→    drawCanvas();
   844→  }, [drawCanvas]);
   845→
   846→  const getMousePos = (e: React.MouseEvent): { x: number; y: number } => {
   847→    const canvas = canvasRef.current;
   848→    if (!canvas) return { x: 0, y: 0 };
   849→    const rect = canvas.getBoundingClientRect();
   850→    return { x: (e.clientX - rect.left) / canvasSize.scale, y: (e.clientY - rect.top) / canvasSize.scale };
   851→  };
   852→
   853→  // Touch position helper for mobile
   854→  const getTouchPos = (e: React.TouchEvent): { x: number; y: number } => {
   855→    const canvas = canvasRef.current;
   856→    if (!canvas || e.touches.length === 0) return { x: 0, y: 0 };
   857→    const rect = canvas.getBoundingClientRect();
   858→    const touch = e.touches[0];
   859→    return { x: (touch.clientX - rect.left) / canvasSize.scale, y: (touch.clientY - rect.top) / canvasSize.scale };
   860→  };
   861→
   862→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {
   863→    for (let i = objects.length - 1; i >= 0; i--) {
   864→      const obj = objects[i];
   865→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   866→        const dist = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   867→        if (dist <= (obj.radius || 50)) return obj;
   868→      }
   869→      if (obj.type === 'arrow') {
   870→        const lineLength = Math.sqrt((obj.endX! - obj.x) ** 2 + (obj.endY! - obj.y) ** 2);
   871→        const d1 = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   872→        const d2 = Math.sqrt((x - obj.endX!) ** 2 + (y - obj.endY!) ** 2);
   873→        if (d1 + d2 <= lineLength + 15) return obj;
   874→      }
   875→      if (obj.type === 'rectangle') {
   876→        const rx = Math.min(obj.x, obj.x + (obj.width || 100));
   877→        const ry = Math.min(obj.y, obj.y + (obj.height || 80));
   878→        const rw = Math.abs(obj.width || 100);
   879→        const rh = Math.abs(obj.height || 80);
   880→        if (x >= rx - 10 && x <= rx + rw + 10 && y >= ry - 10 && y <= ry + rh + 10) return obj;
   881→      }
   882→    }
   883→    return null;
   884→  };
   885→
   886→  const handleMouseDown = (e: React.MouseEvent) => {
   887→    const pos = getMousePos(e);
   888→    if (currentTool === 'select') {
   889→      const obj = findObjectAtPosition(pos.x, pos.y);
   890→      if (obj) {
   891→        setSelectedObjectId(obj.id);
   892→        setIsDragging(true);
   893→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });

   895→        setSelectedObjectId(null);
   896→      }

   898→      setIsDrawingCircle(true);
   899→      setCircleStart(pos);
   900→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
   901→    } else if (currentTool === 'circlePhoto') {
   902→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
   903→      setShowAddPhotoDialog(true);

   905→      setIsDrawingArrow(true);
   906→      setArrowStart(pos);
   907→      setArrowPreview(pos);
   908→    } else if (currentTool === 'rectangle') {
   909→      setIsDrawingRect(true);
   910→      setRectStart(pos);
   911→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
   912→    }
   913→  };
   914→
   915→  const handleMouseMove = (e: React.MouseEvent) => {
   916→    const pos = getMousePos(e);
   917→    if (isDragging && selectedObjectId) {
   918→      setObjects(prev => prev.map(obj => {
   919→        if (obj.id !== selectedObjectId) return obj;
   920→        const newX = pos.x - dragOffset.x;
   921→        const newY = pos.y - dragOffset.y;
   922→        if (obj.type === 'arrow') {
   923→          const dx = obj.endX! - obj.x;
   924→          const dy = obj.endY! - obj.y;
   925→          return { ...obj, x: newX, y: newY, endX: newX + dx, endY: newY + dy };
   926→        }
   927→        if (obj.type === 'rectangle') {
   928→          return { ...obj, x: newX, y: newY };
   929→        }
   930→        return { ...obj, x: newX, y: newY };
   931→      }));
   932→    }
   933→    if (isDrawingArrow) setArrowPreview(pos);
   934→    if (isDrawingCircle && circleStart) {
   935→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   936→      setCirclePreview({ x: circleStart.x, y: circleStart.y, radius: dist });
   937→    }
   938→    if (isDrawingRect && rectStart) {
   939→      setRectPreview({ x: rectStart.x, y: rectStart.y, w: pos.x - rectStart.x, h: pos.y - rectStart.y });
   940→    }
   941→  };
   942→
   943→  const handleMouseUp = (e: React.MouseEvent) => {
   944→    const pos = getMousePos(e);
   945→    if (isDragging) setIsDragging(false);
   946→    if (isDrawingCircle && circleStart && currentTool === 'circle') {
   947→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   948→      const finalRadius = dist > 10 ? dist : circleRadius;
   949→      const newCircle: EditorCircle = { id: `circle-${Date.now()}`, type: 'circle', x: circleStart.x, y: circleStart.y, radius: finalRadius, color: currentColor };
   950→      setObjects(prev => [...prev, newCircle]);
   951→      setSelectedObjectId(newCircle.id);
   952→      setIsDrawingCircle(false);
   953→      setCircleStart(null);
   954→      setCirclePreview(null);
   955→      setCurrentTool('select');
   956→    }
   957→    if (isDrawingRect && rectStart && currentTool === 'rectangle') {
   958→      const w = pos.x - rectStart.x;
   959→      const h = pos.y - rectStart.y;
   960→      if (Math.abs(w) > 10 && Math.abs(h) > 10) {
   961→        const newRect: EditorCircle = { id: `rect-${Date.now()}`, type: 'rectangle', x: rectStart.x, y: rectStart.y, width: w, height: h, color: currentColor };
   962→        setObjects(prev => [...prev, newRect]);
   963→        setSelectedObjectId(newRect.id);
   964→      }
   965→      setIsDrawingRect(false);
   966→      setRectStart(null);
   967→      setRectPreview(null);
   968→      setCurrentTool('select');
   969→    }
   970→    if (isDrawingArrow && arrowStart && currentTool === 'arrow') {
   971→      const distance = Math.sqrt((pos.x - arrowStart.x) ** 2 + (pos.y - arrowStart.y) ** 2);
   972→      if (distance > 10) {
   973→        const newArrow: EditorCircle = { id: `arrow-${Date.now()}`, type: 'arrow', x: arrowStart.x, y: arrowStart.y, endX: pos.x, endY: pos.y, color: currentColor };
   974→        setObjects(prev => [...prev, newArrow]);
   975→        setSelectedObjectId(newArrow.id);
   976→      }
   977→      setIsDrawingArrow(false);
   978→      setArrowStart(null);
   979→      setArrowPreview(null);
   980→      setCurrentTool('select');
   981→    }
   982→  };
   983→
   984→  // Touch event handlers for mobile
   985→  const handleTouchStart = (e: React.TouchEvent) => {
   986→    e.preventDefault();
   987→    const pos = getTouchPos(e);
   988→    if (currentTool === 'select') {
   989→      const obj = findObjectAtPosition(pos.x, pos.y);
   990→      if (obj) {
   991→        setSelectedObjectId(obj.id);
   992→        setIsDragging(true);
   993→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   994→      } else {
   995→        setSelectedObjectId(null);
   996→      }
   997→    } else if (currentTool === 'circle') {
   998→      setIsDrawingCircle(true);
   999→      setCircleStart(pos);
  1000→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
  1001→    } else if (currentTool === 'circlePhoto') {
  1002→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
  1003→      setShowAddPhotoDialog(true);
  1004→    } else if (currentTool === 'arrow') {
  1005→      setIsDrawingArrow(true);
  1006→      setArrowStart(pos);
  1007→      setArrowPreview(pos);
  1008→    } else if (currentTool === 'rectangle') {
  1009→      setIsDrawingRect(true);
  1010→      setRectStart(pos);
  1011→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
  1012→    }
  1013→  };
  1014→

  1301→// Função para gerar imagem preta com número de série
  1302→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1303→  if (!serialNumber || serialNumber.trim() === '') return null;
  1304→  
  1305→  const canvas = document.createElement('canvas');
  1306→  canvas.width = 400;
  1307→  canvas.height = 200;
  1308→  const ctx = canvas.getContext('2d');
  1309→  
  1310→  if (!ctx) return null;
  1311→  
  1312→  // Fundo preto
  1313→  ctx.fillStyle = '#000000';
  1314→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1315→  
  1316→  // Texto branco
  1317→  ctx.fillStyle = '#FFFFFF';
  1318→  ctx.font = 'bold 32px monospace';
  1319→  ctx.textAlign = 'center';
  1320→  ctx.textBaseline = 'middle';
  1321→  
  1322→  // Desenhar o número de série
  1323→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1324→  
  1325→  return canvas.toDataURL('image/png');
  1326→}
