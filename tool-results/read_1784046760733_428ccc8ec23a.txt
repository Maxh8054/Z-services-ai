   898→// Get translation for a key
   899→export function t(key: string, lang: Language, params?: Record<string, string | number>): string {
   900→  // First try to get translation from current language
   901→  let text = translations[lang]?.[key];
   902→  
   903→  // If not found, try Portuguese as fallback
   904→  if (!text) {
   905→    text = translations['pt'][key];
   906→  }
   907→  
   908→  // If still not found, use defaultValue or the key itself
   909→  if (!text) {
   910→    text = (params?.defaultValue as string) || key;
   911→  }
   912→  
   913→  // Replace any parameters in the text
   914→  if (params) {
   915→    Object.entries(params).forEach(([paramKey, value]) => {
   916→      if (paramKey !== 'defaultValue') {
   917→        text = text.replace(`{${paramKey}}`, String(value));
   918→      }
   919→    });
   920→  }
   921→  
   922→  return text;
   923→}

   925→// Translate dynamic text using AI
   926→export async function translateText(text: string, targetLang: Language, sourceLang: Language = 'pt'): Promise<string> {
   927→  if (!text || text.trim() === '') return text;
   928→  if (sourceLang === targetLang) return text;
   929→  
   930→  const langNames: Record<Language, string> = {
   931→    pt: 'Portuguese',
   932→    en: 'English',
   933→    ja: 'Japanese',
   934→    zh: 'Chinese (Mandarin)',
   935→  };
   936→  
   937→  try {
   938→    const response = await fetch('/api/translate', {
   939→      method: 'POST',
   940→      headers: { 'Content-Type': 'application/json' },
   941→      body: JSON.stringify({
   942→        text,
   943→        sourceLang,
   944→        targetLang,
   945→        sourceLangName: langNames[sourceLang],
   946→        targetLangName: langNames[targetLang],
   947→      }),
   948→    });
   949→    
   950→    const data = await response.json();
   951→    return data.translation || text;
   952→  } catch (error) {
   953→    console.error('Translation error:', error);
   954→    return text;
   955→  }
   956→}
   957→
   958→// Batch translate multiple texts
   959→export async function translateBatch(
   960→  texts: string[],
   961→  targetLang: Language,
   962→  sourceLang: Language = 'pt'
   963→): Promise<string[]> {
   964→  if (sourceLang === targetLang) return texts;
   965→  
   966→  const langNames: Record<Language, string> = {
   967→    pt: 'Portuguese',
   968→    en: 'English',
   969→    ja: 'Japanese',
   970→    zh: 'Chinese (Mandarin)',
   971→  };
   972→  
   973→  try {
   974→    const response = await fetch('/api/translate', {
   975→      method: 'POST',
   976→      headers: { 'Content-Type': 'application/json' },
   977→      body: JSON.stringify({
   978→        texts,
   979→        sourceLang,
   980→        targetLang,
   981→        sourceLangName: langNames[sourceLang],
   982→        targetLangName: langNames[targetLang],
   983→        batch: true,
   984→      }),
   985→    });
   986→    
   987→    const data = await response.json();
   988→    return data.translations || texts;
   989→  } catch (error) {
   990→    console.error('Batch translation error:', error);
   991→    return texts;
   992→  }
   993→}
