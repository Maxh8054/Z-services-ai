    51→function sanitizeForFilename(text: string): string {
    52→  return text
    53→    .trim()
    54→    .substring(0, 60)
    55→    .replace(/[<>:"/\\|?*\x00-\x1f]/g, '')
    56→    .replace(/\s+/g, '_')
    57→    .replace(/_+/g, '_')
    58→    .replace(/^_|_$/g, '')
    59→    || 'export';
    60→}

    62→function generateReportId(): string {
    63→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No I, O, 0, 1 to avoid confusion
    64→  let id = 'ZS-';
    65→  for (let i = 0; i < 6; i++) {
    66→    id += chars.charAt(Math.floor(Math.random() * chars.length));
    67→  }
    68→  return id;
    69→}

   100→// Componente de Menu de Ações Mobile
   101→function MobileActionMenu({
   102→  onAddPhoto,
   103→  onDataDialog,
   104→  onExportDialog,
   105→  onEmailDialog,
   106→  onClearConfirm,
   107→  reportData,
   108→  reportType,
   109→  t,
   110→}: {
   111→  onAddPhoto: () => void;
   112→  onDataDialog: () => void;
   113→  onExportDialog: () => void;
   114→  onEmailDialog: () => void;
   115→  onClearConfirm: () => void;
   116→  reportData: unknown;
   117→  reportType: 'home' | 'inspecao';
   118→  t: (key: string, params?: Record<string, string | number>) => string;
   119→}) {
   120→  const [isOpen, setIsOpen] = useState(false);
   121→  const [timeLeft, setTimeLeft] = useState(10);
   122→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   123→  const countdownRef = useRef(10);
   124→
   125→  useEffect(() => {
   126→    if (isOpen) {
   127→      // Reset countdown ref
   128→      countdownRef.current = 10;
   129→      
   130→      timerRef.current = setInterval(() => {
   131→        countdownRef.current -= 1;
   132→        const newTime = countdownRef.current;
   133→        setTimeLeft(newTime);
   134→        
   135→        if (newTime <= 0) {
   136→          setIsOpen(false);
   137→          if (timerRef.current) {
   138→            clearInterval(timerRef.current);
   139→            timerRef.current = null;
   140→          }
   141→        }
   142→      }, 1000);
   143→      
   144→      return () => {
   145→        if (timerRef.current) {
   146→          clearInterval(timerRef.current);
   147→          timerRef.current = null;
   148→        }
   149→      };
   150→    }
   151→  }, [isOpen]);
   152→
   153→  const handleAction = (action: () => void) => {
   154→    action();
   155→    setIsOpen(false);
   156→    if (timerRef.current) {
   157→      clearInterval(timerRef.current);
   158→      timerRef.current = null;
   159→    }
   160→  };
   161→
   162→  return (
   163→    <div className="md:hidden fixed bottom-4 right-4 z-50">
   164→      {/* Menu expandido */}
   165→      {isOpen && (
   166→        <div className="absolute bottom-16 right-0 bg-black/95 backdrop-blur-sm rounded-2xl p-3 shadow-2xl border border-white/10 min-w-[200px]">
   167→          <div className="text-xs text-white/60 mb-2 text-center">
   168→            Fechando em {timeLeft}s
   169→          </div>
   170→          <div className="grid grid-cols-3 gap-2">
   171→            <Button 
   172→              size="icon" 
   173→              className="rounded-full bg-green-600 hover:bg-green-700 h-12 w-12" 
   174→              onClick={() => handleAction(onAddPhoto)}
   175→              title={t('action.addPhoto')}
   176→            >
   177→              <Plus className="h-5 w-5" />
   178→            </Button>
   179→            <Button 
   180→              size="icon" 
   181→              className="rounded-full bg-blue-600 hover:bg-blue-700 h-12 w-12" 
   182→              onClick={() => handleAction(onDataDialog)}
   183→              title={t('action.importExport')}
   184→            >
   185→              <FileDown className="h-5 w-5" />
   186→            </Button>
   187→            <Button 
   188→              size="icon" 
   189→              className="rounded-full bg-purple-600 hover:bg-purple-700 h-12 w-12" 
   190→              onClick={() => handleAction(onExportDialog)}
   191→              title={t('action.download')}
   192→            >
   193→              <Download className="h-5 w-5" />
   194→            </Button>
   195→            <Button 
   196→              size="icon" 
   197→              className="rounded-full bg-orange-500 hover:bg-orange-600 h-12 w-12" 
   198→              onClick={() => handleAction(onEmailDialog)}
   199→              title="Email"
   200→            >
   201→              <Mail className="h-5 w-5" />
   202→            </Button>
   203→            <Button 
   204→              size="icon" 
   205→              className="rounded-full bg-red-600 hover:bg-red-700 h-12 w-12" 
   206→              onClick={() => handleAction(onClearConfirm)}
   207→              title={t('action.clear')}
   208→            >
   209→              <Trash2 className="h-5 w-5" />
   210→            </Button>
   211→          </div>
   212→        </div>
   213→      )}
   214→      
   215→      {/* Botão principal */}
   216→      <Button
   217→        className={`rounded-full h-14 w-14 shadow-lg ${isOpen ? 'bg-red-500 hover:bg-red-600' : 'bg-orange-500 hover:bg-orange-600'}`}
   218→        onClick={() => setIsOpen(!isOpen)}
   219→      >
   220→        {isOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   221→      </Button>
   222→    </div>
   223→  );
   224→}

   368→// Componente para a seção de peças adicionais
   369→function AdditionalPartsSection({
   370→  parentPn,
   371→  photoId,
   372→  additionalParts,
   373→  onAddPart,
   374→  onRemovePart,
   375→  onUpdatePart,
   376→  t,
   377→}: {
   378→  parentPn: string;
   379→  photoId?: string;
   380→  additionalParts: AdditionalPart[];
   381→  onAddPart: (part: AdditionalPart) => void;
   382→  onRemovePart: (id: string) => void;
   383→  onUpdatePart?: (id: string, data: Partial<AdditionalPart>) => void;
   384→  t: (key: string, params?: Record<string, string | number>) => string;
   385→}) {
   386→  const [newPn, setNewPn] = useState('');
   387→  const [newSerialNumber, setNewSerialNumber] = useState('');
   388→  const [newPartName, setNewPartName] = useState('');
   389→  const [newQuantity, setNewQuantity] = useState('');
   390→  const [newCriticality, setNewCriticality] = useState<'Alta' | 'Média' | 'Baixa' | ''>('');
   391→  const [showExcelPaste, setShowExcelPaste] = useState(false);
   392→  const [excelPasteData, setExcelPasteData] = useState('');
   393→
   394→  const photoAdditionalParts = additionalParts.filter(p =>
   395→    (photoId && p.photoId === photoId) || (!photoId && p.parentPn === parentPn)
   396→  );
   397→
   398→  const handleAddPart = () => {
   399→    if (!newPn.trim() || !newPartName.trim() || !newQuantity.trim()) return;
   400→    
   401→    onAddPart({
   402→      id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   403→      pn: newPn.trim(),
   404→      serialNumber: newSerialNumber.trim(),
   405→      partName: newPartName.trim(),
   406→      quantity: newQuantity.trim(),
   407→      criticality: newCriticality,
   408→      parentPn: parentPn,
   409→      photoId: photoId || undefined,
   410→    });
   411→
   412→    setNewPn('');
   413→    setNewSerialNumber('');
   414→    setNewPartName('');
   415→    setNewQuantity('');
   416→    setNewCriticality('');
   417→  };
   418→
   419→  // Processar dados do Excel colados
   420→  const handlePasteExcel = () => {
   421→    if (!excelPasteData.trim()) return;
   422→    
   423→    // Parse dos dados - formato: Part No.      Q'ty    Part name (tab ou espaços)
   424→    const lines = excelPasteData.trim().split('\n');
   425→    
   426→    lines.forEach((line) => {
   427→      // Divide por tab ou múltiplos espaços
   428→      const parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p);
   429→      
   430→      if (parts.length >= 3) {
   431→        const pn = parts[0];
   432→        const quantity = parts[1];
   433→        const partName = parts.slice(2).join(' '); // Nome pode ter espaços
   434→        
   435→        onAddPart({
   436→          id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   437→          pn: pn,
   438→          serialNumber: '',
   439→          partName: partName,
   440→          quantity: quantity,
   441→          criticality: '',
   442→          parentPn: parentPn,
   443→          photoId: photoId || undefined,
   444→        });
   445→      }
   446→    });
   447→    
   448→    setExcelPasteData('');
   449→    setShowExcelPaste(false);
   450→  };
   451→
   452→  // Get badge color based on criticality
   453→  const getCriticalityBadge = (criticality: string) => {
   454→    switch (criticality) {
   455→      case 'Alta':
   456→        return <Badge className="bg-red-500 text-white text-xs h-5 px-2">{t('photo.criticalityHigh')}</Badge>;
   457→      case 'Média':
   458→        return <Badge className="bg-yellow-500 text-white text-xs h-5 px-2">{t('photo.criticalityMedium')}</Badge>;
   459→      case 'Baixa':
   460→        return <Badge className="bg-green-500 text-white text-xs h-5 px-2">{t('photo.criticalityLow')}</Badge>;
   461→      default:
   462→        return null;
   463→    }
   464→  };
   465→
   466→  return (
   467→    <div className="mt-3 p-3 bg-orange-50 rounded-lg border-2 border-orange-400 space-y-2">
   468→      <div className="flex items-center justify-between">
   469→        <Label className="text-sm text-orange-800 font-bold flex items-center gap-2">
   470→          <ListTree className="h-4 w-4" />
   471→          {t('subparts.partsOf', { pn: parentPn || '...' })}
   472→        </Label>
   473→        <Button
   474→          variant="ghost"
   475→          size="sm"
   476→          className="h-7 text-xs text-orange-600 hover:text-orange-800"
   477→          onClick={() => setShowExcelPaste(!showExcelPaste)}
   478→        >
   479→          {showExcelPaste ? 'Manual' : 'Colar Excel'}
   480→        </Button>
   481→      </div>
   482→      
   483→      {/* Excel Paste Mode */}
   484→      {showExcelPaste && (
   485→        <div className="space-y-2">
   486→          <p className="text-xs text-orange-700">
   487→            Cole os dados do Excel no formato: Part No. | Q'ty | Part name (uma linha por peça)
   488→          </p>
   489→          <Textarea
   490→            className="min-h-[80px] text-sm"
   491→            placeholder="4244836&#9;4&#9;BUCHA&#10;4244837&#9;2&#9;ANEL"
   492→            value={excelPasteData}
   493→            onChange={(e) => setExcelPasteData(e.target.value)}
   494→          />
   495→          <div className="flex gap-2">
   496→            <Button
   497→              size="sm"
   498→              className="bg-orange-500 hover:bg-orange-600"
   499→              onClick={handlePasteExcel}
   500→              disabled={!excelPasteData.trim()}
   501→            >
   502→              <Plus className="h-4 w-4 mr-1" /> Adicionar
   503→            </Button>
   504→            <Button
   505→              variant="outline"
   506→              size="sm"
   507→              onClick={() => { setExcelPasteData(''); setShowExcelPaste(false); }}
   508→            >
   509→              Cancelar
   510→            </Button>
   511→          </div>
   512→        </div>
   513→      )}
   514→      
   515→      {/* Manual Mode */}
   516→      {!showExcelPaste && (
   517→        <>
   518→      
   519→      {photoAdditionalParts.length > 0 && (
   520→        <div className="space-y-1 max-h-32 overflow-y-auto">
   521→          {photoAdditionalParts.map((part) => (
   522→            <div key={part.id} className="flex items-center gap-2 bg-white p-2 rounded border text-sm">
   523→              <span className="text-orange-600 font-mono font-bold">└─</span>
   524→              <span className="font-semibold text-gray-700">{part.pn}</span>
   525→              {part.serialNumber && (
   526→                <span className="text-gray-500">({part.serialNumber})</span>
   527→              )}
   528→              <span className="text-gray-600 flex-1 truncate">{part.partName}</span>
   529→              <Badge variant="secondary" className="text-xs h-5 px-2">{t('photo.quantity')}: {part.quantity}</Badge>
   530→              {getCriticalityBadge(part.criticality)}
   531→              <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 min-h-[44px] min-w-[44px] md:min-h-0 md:min-w-0" onClick={() => onRemovePart(part.id)}>
   532→                <X className="h-4 w-4" />
   533→              </Button>
   534→            </div>
   535→          ))}
   536→        </div>
   537→      )}
   538→
   539→      {/* Mobile layout - stacked */}
   540→      <div className="md:hidden space-y-2">
   541→        <div className="grid grid-cols-2 gap-2">
   542→          <Input className="h-11 text-base" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />
   543→          <Input className="h-11 text-base" placeholder={t('photo.serialPlaceholder')} value={newSerialNumber} onChange={(e) => setNewSerialNumber(e.target.value)} />
   544→        </div>
   545→        <Input className="h-11 text-base" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   546→        <div className="grid grid-cols-2 gap-2">
   547→          <Input className="h-11 text-base" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />
   548→          <Select value={newCriticality} onValueChange={(v) => setNewCriticality(v as 'Alta' | 'Média' | 'Baixa' | '')}>
   549→            <SelectTrigger className="h-11 text-base">
   550→              <SelectValue placeholder={t('photo.criticality')} />

   571→          <SelectTrigger className="h-8 text-xs col-span-2">

   577→            <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>

   580→        <Button size="icon" className="h-8 w-8 bg-orange-500 hover:bg-orange-600 col-span-2" onClick={handleAddPart} disabled={!newPn.trim() || !newPartName.trim() || !newQuantity.trim()}>
   581→          <Plus className="h-4 w-4" />
   582→        </Button>
   583→      </div>
   584→        </>
   585→      )}
   586→    </div>
   587→  );
   588→}

   593→// Editor de Foto
   594→function PhotoEditor({
   595→  photo,
   596→  editMode = 'primary',
   597→  onSave,
   598→  onClose,
   599→  t,
   600→}: {
   601→  photo: PhotoData;
   602→  editMode?: PhotoEditMode;
   603→  onSave: (data: Partial<PhotoData>) => void;
   604→  onClose: () => void;
   605→  t: (key: string, params?: Record<string, string | number>) => string;
   606→}) {
   607→  const canvasRef = useRef<HTMLCanvasElement>(null);
   608→  const containerRef = useRef<HTMLDivElement>(null);
   609→  const imgRef = useRef<HTMLImageElement | null>(null);
   610→  
   611→  const [objects, setObjects] = useState<EditorCircle[]>(() => {
   612→    return (photo.embeddedPhotos || []).map(ep => ({
   613→      id: ep.id,
   614→      type: 'circleWithPhoto' as const,
   615→      x: ep.circleX,
   616→      y: ep.circleY,
   617→      radius: ep.circleRadius,
   618→      color: '#f97316',
   619→      imageData: ep.imageData,
   620→    }));
   621→  });
   622→  
   623→  const [currentTool, setCurrentTool] = useState<'select' | 'circle' | 'circlePhoto' | 'arrow' | 'rectangle'>('select');
   624→  const [selectedObjectId, setSelectedObjectId] = useState<string | null>(null);
   625→  const [isDragging, setIsDragging] = useState(false);
   626→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 });
   627→  const [currentColor, setCurrentColor] = useState('#f97316');
   628→  const [circleRadius, setCircleRadius] = useState(80);
   629→  const [showAddPhotoDialog, setShowAddPhotoDialog] = useState(false);
   630→  const [pendingPhotoCircle, setPendingPhotoCircle] = useState<EditorCircle | null>(null);
   631→  const [isDrawingArrow, setIsDrawingArrow] = useState(false);
   632→  const [arrowStart, setArrowStart] = useState<{ x: number; y: number } | null>(null);
   633→  const [arrowPreview, setArrowPreview] = useState<{ x: number; y: number } | null>(null);
   634→  const [isDrawingCircle, setIsDrawingCircle] = useState(false);
   635→  const [circleStart, setCircleStart] = useState<{ x: number; y: number } | null>(null);
   636→  const [circlePreview, setCirclePreview] = useState<{ x: number; y: number; radius: number } | null>(null);
   637→  const [isDrawingRect, setIsDrawingRect] = useState(false);
   638→  const [rectStart, setRectStart] = useState<{ x: number; y: number } | null>(null);
   639→  const [rectPreview, setRectPreview] = useState<{ x: number; y: number; w: number; h: number } | null>(null);
   640→  const [rotation, setRotation] = useState(0);
   641→
   642→  // Determinar qual imagem editar baseado no modo
   643→  const getActiveImageSrc = () => {
   644→    if (editMode === 'secondary') {
   645→      return photo.secondaryImageData;
   646→    }
   647→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   648→      // Criar imagem combinada lado a lado
   649→      return null; // Será tratado separadamente
   650→    }
   651→    return photo.editedImageData || photo.imageData;
   652→  };
   653→  
   654→  const imageSrc = getActiveImageSrc();
   655→  const [dualCanvasSize, setDualCanvasSize] = useState<{ width: number; height: number } | null>(null);
   656→  const [canvasSize, setCanvasSize] = useState({ width: 0, height: 0, scale: 1 });
   657→
   658→  useEffect(() => {
   659→    const container = containerRef.current;
   660→    if (!container) return;
   661→    const containerWidth = Math.min(container.clientWidth - 32, 900);
   662→    const containerHeight = 500;
   663→
   664→    // Modo duplo: carregar ambas as imagens lado a lado
   665→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   666→      const img1 = new window.Image();
   667→      const img2 = new window.Image();
   668→      
   669→      let loaded = 0;
   670→      const checkLoaded = () => {
   671→        loaded++;
   672→        if (loaded === 2) {
   673→          // Calcular dimensões para ambas as imagens lado a lado
   674→          const halfWidth = containerWidth / 2 - 10;
   675→          const scale1 = Math.min(halfWidth / img1.width, containerHeight / img1.height);
   676→          const scale2 = Math.min(halfWidth / img2.width, containerHeight / img2.height);
   677→          const scale = Math.min(scale1, scale2);
   678→          
   679→          const totalWidth = (img1.width + img2.width) * scale + 10;
   680→          const totalHeight = Math.max(img1.height, img2.height) * scale;
   681→          
   682→          setCanvasSize({ width: totalWidth, height: totalHeight, scale });
   683→          setDualCanvasSize({ width: img1.width * scale, height: img1.height * scale });
   684→          
   685→          // Criar imagem combinada temporária
   686→          const tempCanvas = document.createElement('canvas');
   687→          tempCanvas.width = img1.width + img2.width;
   688→          tempCanvas.height = Math.max(img1.height, img2.height);
   689→          const ctx = tempCanvas.getContext('2d');
   690→          if (ctx) {
   691→            ctx.drawImage(img1, 0, 0);
   692→            ctx.drawImage(img2, img1.width, 0);
   693→          }
   694→          
   695→          const combinedImg = new window.Image();
   696→          combinedImg.onload = () => {
   697→            imgRef.current = combinedImg;
   698→          };
   699→          combinedImg.src = tempCanvas.toDataURL('image/png');
   700→        }
   701→      };
   702→      
   703→      img1.onload = checkLoaded;
   704→      img2.onload = checkLoaded;
   705→      img1.src = photo.imageData;
   706→      img2.src = photo.secondaryImageData;
   707→      return;
   708→    }
   709→
   710→    // Modo normal (primário ou secundário)
   711→    if (!imageSrc) return;
   712→    const img = new window.Image();
   713→    img.onload = () => {
   714→      imgRef.current = img;
   715→      let imgWidth = img.width;
   716→      let imgHeight = img.height;
   717→      // Swap dimensions for 90/270 degree rotation
   718→      if (rotation === 90 || rotation === 270) {
   719→        [imgWidth, imgHeight] = [imgHeight, imgWidth];
   720→      }
   721→      const scale = Math.min(containerWidth / imgWidth, containerHeight / imgHeight);
   722→      setCanvasSize({ width: imgWidth * scale, height: imgHeight * scale, scale });
   723→    };
   724→    img.src = imageSrc;
   725→  }, [imageSrc, editMode, photo.imageData, photo.secondaryImageData, rotation]);
   726→
   727→  const drawCanvas = useCallback(() => {
   728→    const canvas = canvasRef.current;
   729→    const img = imgRef.current;
   730→    if (!canvas || !img || !imageSrc) return;
   731→
   732→    const ctx = canvas.getContext('2d');
   733→    if (!ctx) return;
   734→
   735→    const { width, height, scale } = canvasSize;
   736→    canvas.width = width;
   737→    canvas.height = height;
   738→    ctx.save();
   739→    if (rotation > 0) {
   740→      ctx.translate(width / 2, height / 2);
   741→      ctx.rotate((rotation * Math.PI) / 180);
   742→      ctx.translate(-width / 2, -height / 2);
   743→    }
   744→    ctx.drawImage(img, 0, 0, width, height);
   745→    ctx.restore();
   746→
   747→    objects.forEach((obj) => {
   748→      const isSelected = obj.id === selectedObjectId;
   749→      
   750→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   751→        const x = obj.x * scale;
   752→        const y = obj.y * scale;
   753→        const radius = (obj.radius || 50) * scale;
   754→
   755→        if (obj.type === 'circleWithPhoto' && obj.imageData) {
   756→          const embeddedImg = new window.Image();
   757→          embeddedImg.onload = () => {
   758→            ctx.save();
   759→            ctx.beginPath();
   760→            ctx.arc(x, y, radius - 2, 0, Math.PI * 2);
   761→            ctx.closePath();
   762→            ctx.clip();
   763→            ctx.drawImage(embeddedImg, x - radius + 2, y - radius + 2, (radius - 2) * 2, (radius - 2) * 2);
   764→            ctx.restore();
   765→            ctx.beginPath();
   766→            ctx.arc(x, y, radius, 0, Math.PI * 2);
   767→            ctx.strokeStyle = obj.color;
   768→            ctx.lineWidth = isSelected ? 4 : 3;
   769→            ctx.stroke();
   770→          };
   771→          embeddedImg.src = obj.imageData;
   772→        } else {
   773→          ctx.beginPath();
   774→          ctx.arc(x, y, radius, 0, Math.PI * 2);
   775→          ctx.strokeStyle = obj.color;
   776→          ctx.lineWidth = isSelected ? 4 : 3;
   777→          ctx.stroke();
   778→        }
   779→      }
   780→      
   781→      if (obj.type === 'arrow') {
   782→        const startX = obj.x * scale;
   783→        const startY = obj.y * scale;
   784→        const endX = (obj.endX || obj.x + 50) * scale;
   785→        const endY = (obj.endY || obj.y + 50) * scale;
   786→
   787→        ctx.beginPath();
   788→        ctx.moveTo(startX, startY);
   789→        ctx.lineTo(endX, endY);
   790→        ctx.strokeStyle = obj.color;
   791→        ctx.lineWidth = isSelected ? 4 : 3;
   792→        ctx.stroke();
   793→
   794→        const angle = Math.atan2(endY - startY, endX - startX);
   795→        ctx.beginPath();
   796→        ctx.moveTo(endX, endY);
   797→        ctx.lineTo(endX - 15 * Math.cos(angle - Math.PI / 6), endY - 15 * Math.sin(angle - Math.PI / 6));
   798→        ctx.moveTo(endX, endY);
   799→        ctx.lineTo(endX - 15 * Math.cos(angle + Math.PI / 6), endY - 15 * Math.sin(angle + Math.PI / 6));
   800→        ctx.stroke();
   801→      }
   802→    });
   803→
   804→    // Rectangle objects
   805→    objects.forEach((obj) => {
   806→      if (obj.type === 'rectangle') {
   807→        const isSelected = obj.id === selectedObjectId;
   808→        const rx = obj.x * scale;
   809→        const ry = obj.y * scale;
   810→        const rw = (obj.width || 100) * scale;
   811→        const rh = (obj.height || 80) * scale;
   812→        ctx.strokeStyle = obj.color;
   813→        ctx.lineWidth = isSelected ? 4 : 3;
   814→        ctx.strokeRect(rx, ry, rw, rh);
   815→      }
   816→    });
   817→
   818→    if (isDrawingArrow && arrowStart && arrowPreview) {
   819→      ctx.beginPath();
   820→      ctx.moveTo(arrowStart.x * scale, arrowStart.y * scale);
   821→      ctx.lineTo(arrowPreview.x * scale, arrowPreview.y * scale);
   822→      ctx.strokeStyle = currentColor;
   823→      ctx.lineWidth = 3;
   824→      ctx.setLineDash([5, 5]);
   825→      ctx.stroke();
   826→      ctx.setLineDash([]);
   827→    }
   828→
   829→    // Circle drag preview
   830→    if (isDrawingCircle && circleStart && circlePreview) {
   831→      ctx.beginPath();
   832→      ctx.arc(circlePreview.x * scale, circlePreview.y * scale, circlePreview.radius * scale, 0, Math.PI * 2);
   833→      ctx.strokeStyle = currentColor;
   834→      ctx.lineWidth = 3;
   835→      ctx.setLineDash([5, 5]);
   836→      ctx.stroke();
   837→      ctx.setLineDash([]);
   838→    }
   839→
   840→    // Rectangle drag preview
   841→    if (isDrawingRect && rectStart && rectPreview) {
   842→      ctx.strokeStyle = currentColor;
   843→      ctx.lineWidth = 3;
   844→      ctx.setLineDash([5, 5]);
   845→      ctx.strokeRect(rectPreview.x * scale, rectPreview.y * scale, rectPreview.w * scale, rectPreview.h * scale);
   846→      ctx.setLineDash([]);
   847→    }
   848→  }, [imageSrc, canvasSize, objects, selectedObjectId, isDrawingArrow, arrowStart, arrowPreview, currentColor, rotation, isDrawingCircle, circleStart, circlePreview, isDrawingRect, rectStart, rectPreview]);
   849→
   850→  useEffect(() => {
   851→    drawCanvas();
   852→  }, [drawCanvas]);
   853→
   854→  const getMousePos = (e: React.MouseEvent): { x: number; y: number } => {
   855→    const canvas = canvasRef.current;
   856→    if (!canvas) return { x: 0, y: 0 };
   857→    const rect = canvas.getBoundingClientRect();
   858→    return { x: (e.clientX - rect.left) / canvasSize.scale, y: (e.clientY - rect.top) / canvasSize.scale };
   859→  };
   860→
   861→  // Touch position helper for mobile
   862→  const getTouchPos = (e: React.TouchEvent): { x: number; y: number } => {
   863→    const canvas = canvasRef.current;
   864→    if (!canvas || e.touches.length === 0) return { x: 0, y: 0 };
   865→    const rect = canvas.getBoundingClientRect();
   866→    const touch = e.touches[0];
   867→    return { x: (touch.clientX - rect.left) / canvasSize.scale, y: (touch.clientY - rect.top) / canvasSize.scale };
   868→  };
   869→
   870→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {
   871→    for (let i = objects.length - 1; i >= 0; i--) {
   872→      const obj = objects[i];
   873→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   874→        const dist = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   875→        if (dist <= (obj.radius || 50)) return obj;
   876→      }
   877→      if (obj.type === 'arrow') {
   878→        const lineLength = Math.sqrt((obj.endX! - obj.x) ** 2 + (obj.endY! - obj.y) ** 2);
   879→        const d1 = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   880→        const d2 = Math.sqrt((x - obj.endX!) ** 2 + (y - obj.endY!) ** 2);
   881→        if (d1 + d2 <= lineLength + 15) return obj;
   882→      }
   883→      if (obj.type === 'rectangle') {
   884→        const rx = Math.min(obj.x, obj.x + (obj.width || 100));
   885→        const ry = Math.min(obj.y, obj.y + (obj.height || 80));
   886→        const rw = Math.abs(obj.width || 100);
   887→        const rh = Math.abs(obj.height || 80);
   888→        if (x >= rx - 10 && x <= rx + rw + 10 && y >= ry - 10 && y <= ry + rh + 10) return obj;
   889→      }
   890→    }
   891→    return null;
   892→  };
   893→
   894→  const handleMouseDown = (e: React.MouseEvent) => {
   895→    const pos = getMousePos(e);
   896→    if (currentTool === 'select') {
   897→      const obj = findObjectAtPosition(pos.x, pos.y);
   898→      if (obj) {
   899→        setSelectedObjectId(obj.id);
   900→        setIsDragging(true);
   901→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   902→      } else {
   903→        setSelectedObjectId(null);
   904→      }
   905→    } else if (currentTool === 'circle') {
   906→      setIsDrawingCircle(true);
   907→      setCircleStart(pos);
   908→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
   909→    } else if (currentTool === 'circlePhoto') {
   910→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
   911→      setShowAddPhotoDialog(true);
   912→    } else if (currentTool === 'arrow') {
   913→      setIsDrawingArrow(true);
   914→      setArrowStart(pos);
   915→      setArrowPreview(pos);
   916→    } else if (currentTool === 'rectangle') {
   917→      setIsDrawingRect(true);
   918→      setRectStart(pos);
   919→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
   920→    }
   921→  };
   922→
   923→  const handleMouseMove = (e: React.MouseEvent) => {
   924→    const pos = getMousePos(e);
   925→    if (isDragging && selectedObjectId) {
   926→      setObjects(prev => prev.map(obj => {
   927→        if (obj.id !== selectedObjectId) return obj;
   928→        const newX = pos.x - dragOffset.x;
   929→        const newY = pos.y - dragOffset.y;
   930→        if (obj.type === 'arrow') {
   931→          const dx = obj.endX! - obj.x;
   932→          const dy = obj.endY! - obj.y;
   933→          return { ...obj, x: newX, y: newY, endX: newX + dx, endY: newY + dy };
   934→        }
   935→        if (obj.type === 'rectangle') {
   936→          return { ...obj, x: newX, y: newY };
   937→        }
   938→        return { ...obj, x: newX, y: newY };
   939→      }));
   940→    }
   941→    if (isDrawingArrow) setArrowPreview(pos);
   942→    if (isDrawingCircle && circleStart) {
   943→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   944→      setCirclePreview({ x: circleStart.x, y: circleStart.y, radius: dist });
   945→    }
   946→    if (isDrawingRect && rectStart) {
   947→      setRectPreview({ x: rectStart.x, y: rectStart.y, w: pos.x - rectStart.x, h: pos.y - rectStart.y });
   948→    }
   949→  };
   950→
   951→  const handleMouseUp = (e: React.MouseEvent) => {
   952→    const pos = getMousePos(e);
   953→    if (isDragging) setIsDragging(false);
   954→    if (isDrawingCircle && circleStart && currentTool === 'circle') {
   955→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   956→      const finalRadius = dist > 10 ? dist : circleRadius;
   957→      const newCircle: EditorCircle = { id: `circle-${Date.now()}`, type: 'circle', x: circleStart.x, y: circleStart.y, radius: finalRadius, color: currentColor };
   958→      setObjects(prev => [...prev, newCircle]);
   959→      setSelectedObjectId(newCircle.id);
   960→      setIsDrawingCircle(false);
   961→      setCircleStart(null);
   962→      setCirclePreview(null);
   963→      setCurrentTool('select');
   964→    }
   965→    if (isDrawingRect && rectStart && currentTool === 'rectangle') {
   966→      const w = pos.x - rectStart.x;
   967→      const h = pos.y - rectStart.y;
   968→      if (Math.abs(w) > 10 && Math.abs(h) > 10) {
   969→        const newRect: EditorCircle = { id: `rect-${Date.now()}`, type: 'rectangle', x: rectStart.x, y: rectStart.y, width: w, height: h, color: currentColor };
   970→        setObjects(prev => [...prev, newRect]);
   971→        setSelectedObjectId(newRect.id);
   972→      }
   973→      setIsDrawingRect(false);
   974→      setRectStart(null);
   975→      setRectPreview(null);
   976→      setCurrentTool('select');
   977→    }
   978→    if (isDrawingArrow && arrowStart && currentTool === 'arrow') {
   979→      const distance = Math.sqrt((pos.x - arrowStart.x) ** 2 + (pos.y - arrowStart.y) ** 2);
   980→      if (distance > 10) {
   981→        const newArrow: EditorCircle = { id: `arrow-${Date.now()}`, type: 'arrow', x: arrowStart.x, y: arrowStart.y, endX: pos.x, endY: pos.y, color: currentColor };
   982→        setObjects(prev => [...prev, newArrow]);
   983→        setSelectedObjectId(newArrow.id);
   984→      }
   985→      setIsDrawingArrow(false);
   986→      setArrowStart(null);
   987→      setArrowPreview(null);
   988→      setCurrentTool('select');
   989→    }
   990→  };
   991→
   992→  // Touch event handlers for mobile
   993→  const handleTouchStart = (e: React.TouchEvent) => {
   994→    e.preventDefault();
   995→    const pos = getTouchPos(e);
   996→    if (currentTool === 'select') {
   997→      const obj = findObjectAtPosition(pos.x, pos.y);
   998→      if (obj) {
   999→        setSelectedObjectId(obj.id);
  1000→        setIsDragging(true);
  1001→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
  1002→      } else {
  1003→        setSelectedObjectId(null);
  1004→      }
  1005→    } else if (currentTool === 'circle') {
  1006→      setIsDrawingCircle(true);
  1007→      setCircleStart(pos);
  1008→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
  1009→    } else if (currentTool === 'circlePhoto') {
  1010→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
  1011→      setShowAddPhotoDialog(true);
  1012→    } else if (currentTool === 'arrow') {

  1309→// Função para gerar imagem preta com número de série
  1310→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1311→  if (!serialNumber || serialNumber.trim() === '') return null;
  1312→  
  1313→  const canvas = document.createElement('canvas');
  1314→  canvas.width = 400;
  1315→  canvas.height = 200;
  1316→  const ctx = canvas.getContext('2d');
  1317→  
  1318→  if (!ctx) return null;
  1319→  
  1320→  // Fundo preto
  1321→  ctx.fillStyle = '#000000';
  1322→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1323→  
  1324→  // Texto branco
  1325→  ctx.fillStyle = '#FFFFFF';
  1326→  ctx.font = 'bold 32px monospace';
  1327→  ctx.textAlign = 'center';
  1328→  ctx.textBaseline = 'middle';
  1329→  
  1330→  // Desenhar o número de série
  1331→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1332→  
  1333→  return canvas.toDataURL('image/png');
  1334→}
