    51→function sanitizeForFilename(text: string): string {
    52→  return text
    53→    .trim()
    54→    .substring(0, 60)
    55→    .replace(/[<>:"/\\|?*\x00-\x1f]/g, '')
    56→    .replace(/\s+/g, '_')
    57→    .replace(/_+/g, '_')
    58→    .replace(/^_|_$/g, '')
    59→    || 'export';
    60→}

    62→function generateReportId(): string {
    63→  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // No I, O, 0, 1 to avoid confusion
    64→  let id = 'ZS-';
    65→  for (let i = 0; i < 6; i++) {
    66→    id += chars.charAt(Math.floor(Math.random() * chars.length));
    67→  }
    68→  return id;
    69→}

   100→// Componente de Menu de Ações Mobile
   101→function MobileActionMenu({
   102→  onAddPhoto,
   103→  onDataDialog,
   104→  onExportDialog,
   105→  onEmailDialog,
   106→  onClearConfirm,
   107→  reportData,
   108→  reportType,
   109→  t,
   110→}: {
   111→  onAddPhoto: () => void;
   112→  onDataDialog: () => void;
   113→  onExportDialog: () => void;
   114→  onEmailDialog: () => void;
   115→  onClearConfirm: () => void;
   116→  reportData: unknown;
   117→  reportType: 'home' | 'inspecao';
   118→  t: (key: string, params?: Record<string, string | number>) => string;
   119→}) {
   120→  const [isOpen, setIsOpen] = useState(false);
   121→  const [timeLeft, setTimeLeft] = useState(10);
   122→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   123→  const countdownRef = useRef(10);
   124→
   125→  useEffect(() => {
   126→    if (isOpen) {
   127→      // Reset countdown ref
   128→      countdownRef.current = 10;
   129→      
   130→      timerRef.current = setInterval(() => {
   131→        countdownRef.current -= 1;
   132→        const newTime = countdownRef.current;
   133→        setTimeLeft(newTime);
   134→        
   135→        if (newTime <= 0) {
   136→          setIsOpen(false);
   137→          if (timerRef.current) {
   138→            clearInterval(timerRef.current);
   139→            timerRef.current = null;
   140→          }
   141→        }
   142→      }, 1000);
   143→      
   144→      return () => {
   145→        if (timerRef.current) {
   146→          clearInterval(timerRef.current);
   147→          timerRef.current = null;
   148→        }
   149→      };
   150→    }
   151→  }, [isOpen]);
   152→
   153→  const handleAction = (action: () => void) => {
   154→    action();
   155→    setIsOpen(false);
   156→    if (timerRef.current) {
   157→      clearInterval(timerRef.current);
   158→      timerRef.current = null;
   159→    }
   160→  };
   161→
   162→  return (
   163→    <div className="md:hidden fixed bottom-4 right-4 z-50">
   164→      {/* Menu expandido */}
   165→      {isOpen && (
   166→        <div className="absolute bottom-16 right-0 bg-black/95 backdrop-blur-sm rounded-2xl p-3 shadow-2xl border border-white/10 min-w-[200px]">
   167→          <div className="text-xs text-white/60 mb-2 text-center">
   168→            Fechando em {timeLeft}s
   169→          </div>
   170→          <div className="grid grid-cols-3 gap-2">
   171→            <Button 
   172→              size="icon" 
   173→              className="rounded-full bg-green-600 hover:bg-green-700 h-12 w-12" 
   174→              onClick={() => handleAction(onAddPhoto)}
   175→              title={t('action.addPhoto')}
   176→            >
   177→              <Plus className="h-5 w-5" />
   178→            </Button>
   179→            <Button 
   180→              size="icon" 
   181→              className="rounded-full bg-blue-600 hover:bg-blue-700 h-12 w-12" 
   182→              onClick={() => handleAction(onDataDialog)}
   183→              title={t('action.importExport')}
   184→            >
   185→              <FileDown className="h-5 w-5" />
   186→            </Button>
   187→            <Button 
   188→              size="icon" 
   189→              className="rounded-full bg-purple-600 hover:bg-purple-700 h-12 w-12" 
   190→              onClick={() => handleAction(onExportDialog)}
   191→              title={t('action.download')}
   192→            >
   193→              <Download className="h-5 w-5" />
   194→            </Button>
   195→            <Button 
   196→              size="icon" 
   197→              className="rounded-full bg-orange-500 hover:bg-orange-600 h-12 w-12" 
   198→              onClick={() => handleAction(onEmailDialog)}
   199→              title="Email"
   200→            >
   201→              <Mail className="h-5 w-5" />
   202→            </Button>
   203→            <Button 
   204→              size="icon" 
   205→              className="rounded-full bg-red-600 hover:bg-red-700 h-12 w-12" 
   206→              onClick={() => handleAction(onClearConfirm)}
   207→              title={t('action.clear')}
   208→            >
   209→              <Trash2 className="h-5 w-5" />
   210→            </Button>
   211→          </div>
   212→        </div>
   213→      )}
   214→      
   215→      {/* Botão principal */}
   216→      <Button
   217→        className={`rounded-full h-14 w-14 shadow-lg ${isOpen ? 'bg-red-500 hover:bg-red-600' : 'bg-orange-500 hover:bg-orange-600'}`}
   218→        onClick={() => setIsOpen(!isOpen)}
   219→      >
   220→        {isOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   221→      </Button>
   222→    </div>
   223→  );
   224→}

   368→// Componente para a seção de peças adicionais
   369→function AdditionalPartsSection({
   370→  parentPn,
   371→  additionalParts,
   372→  onAddPart,
   373→  onRemovePart,
   374→  onUpdatePart,
   375→  t,
   376→}: {
   377→  parentPn: string;
   378→  additionalParts: AdditionalPart[];
   379→  onAddPart: (part: AdditionalPart) => void;
   380→  onRemovePart: (id: string) => void;
   381→  onUpdatePart?: (id: string, data: Partial<AdditionalPart>) => void;
   382→  t: (key: string, params?: Record<string, string | number>) => string;
   383→}) {
   384→  const [newPn, setNewPn] = useState('');
   385→  const [newSerialNumber, setNewSerialNumber] = useState('');
   386→  const [newPartName, setNewPartName] = useState('');
   387→  const [newQuantity, setNewQuantity] = useState('');
   388→  const [newCriticality, setNewCriticality] = useState<'Alta' | 'Média' | 'Baixa' | ''>('');
   389→  const [showExcelPaste, setShowExcelPaste] = useState(false);
   390→  const [excelPasteData, setExcelPasteData] = useState('');
   391→
   392→  const photoAdditionalParts = additionalParts.filter(p => p.parentPn === parentPn);
   393→
   394→  const handleAddPart = () => {
   395→    if (!newPn.trim() || !newPartName.trim() || !newQuantity.trim()) return;
   396→    
   397→    onAddPart({
   398→      id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   399→      pn: newPn.trim(),
   400→      serialNumber: newSerialNumber.trim(),
   401→      partName: newPartName.trim(),
   402→      quantity: newQuantity.trim(),
   403→      criticality: newCriticality,
   404→      parentPn: parentPn,
   405→    });
   406→
   407→    setNewPn('');
   408→    setNewSerialNumber('');
   409→    setNewPartName('');
   410→    setNewQuantity('');
   411→    setNewCriticality('');
   412→  };
   413→
   414→  // Processar dados do Excel colados
   415→  const handlePasteExcel = () => {
   416→    if (!excelPasteData.trim()) return;
   417→    
   418→    // Parse dos dados - formato: Part No.      Q'ty    Part name (tab ou espaços)
   419→    const lines = excelPasteData.trim().split('\n');
   420→    
   421→    lines.forEach((line) => {
   422→      // Divide por tab ou múltiplos espaços
   423→      const parts = line.split(/\t+|\s{2,}/).map(p => p.trim()).filter(p => p);
   424→      
   425→      if (parts.length >= 3) {
   426→        const pn = parts[0];
   427→        const quantity = parts[1];
   428→        const partName = parts.slice(2).join(' '); // Nome pode ter espaços
   429→        
   430→        onAddPart({
   431→          id: `add-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   432→          pn: pn,
   433→          serialNumber: '',
   434→          partName: partName,
   435→          quantity: quantity,
   436→          criticality: '',
   437→          parentPn: parentPn,
   438→        });
   439→      }
   440→    });
   441→    
   442→    setExcelPasteData('');
   443→    setShowExcelPaste(false);
   444→  };
   445→
   446→  // Get badge color based on criticality
   447→  const getCriticalityBadge = (criticality: string) => {
   448→    switch (criticality) {
   449→      case 'Alta':
   450→        return <Badge className="bg-red-500 text-white text-xs h-5 px-2">{t('photo.criticalityHigh')}</Badge>;
   451→      case 'Média':
   452→        return <Badge className="bg-yellow-500 text-white text-xs h-5 px-2">{t('photo.criticalityMedium')}</Badge>;
   453→      case 'Baixa':
   454→        return <Badge className="bg-green-500 text-white text-xs h-5 px-2">{t('photo.criticalityLow')}</Badge>;
   455→      default:
   456→        return null;
   457→    }
   458→  };
   459→
   460→  return (
   461→    <div className="mt-3 p-3 bg-orange-50 rounded-lg border-2 border-orange-400 space-y-2">
   462→      <div className="flex items-center justify-between">
   463→        <Label className="text-sm text-orange-800 font-bold flex items-center gap-2">
   464→          <ListTree className="h-4 w-4" />
   465→          {t('subparts.partsOf', { pn: parentPn || '...' })}
   466→        </Label>
   467→        <Button
   468→          variant="ghost"
   469→          size="sm"
   470→          className="h-7 text-xs text-orange-600 hover:text-orange-800"
   471→          onClick={() => setShowExcelPaste(!showExcelPaste)}
   472→        >
   473→          {showExcelPaste ? 'Manual' : 'Colar Excel'}
   474→        </Button>
   475→      </div>
   476→      
   477→      {/* Excel Paste Mode */}
   478→      {showExcelPaste && (
   479→        <div className="space-y-2">
   480→          <p className="text-xs text-orange-700">
   481→            Cole os dados do Excel no formato: Part No. | Q'ty | Part name (uma linha por peça)
   482→          </p>
   483→          <Textarea
   484→            className="min-h-[80px] text-sm"
   485→            placeholder="4244836&#9;4&#9;BUCHA&#10;4244837&#9;2&#9;ANEL"
   486→            value={excelPasteData}
   487→            onChange={(e) => setExcelPasteData(e.target.value)}
   488→          />
   489→          <div className="flex gap-2">
   490→            <Button
   491→              size="sm"
   492→              className="bg-orange-500 hover:bg-orange-600"
   493→              onClick={handlePasteExcel}
   494→              disabled={!excelPasteData.trim()}
   495→            >
   496→              <Plus className="h-4 w-4 mr-1" /> Adicionar
   497→            </Button>
   498→            <Button
   499→              variant="outline"
   500→              size="sm"
   501→              onClick={() => { setExcelPasteData(''); setShowExcelPaste(false); }}
   502→            >
   503→              Cancelar
   504→            </Button>
   505→          </div>
   506→        </div>
   507→      )}
   508→      
   509→      {/* Manual Mode */}
   510→      {!showExcelPaste && (
   511→        <>
   512→      
   513→      {photoAdditionalParts.length > 0 && (
   514→        <div className="space-y-1 max-h-32 overflow-y-auto">
   515→          {photoAdditionalParts.map((part) => (
   516→            <div key={part.id} className="flex items-center gap-2 bg-white p-2 rounded border text-sm">
   517→              <span className="text-orange-600 font-mono font-bold">└─</span>
   518→              <span className="font-semibold text-gray-700">{part.pn}</span>
   519→              {part.serialNumber && (
   520→                <span className="text-gray-500">({part.serialNumber})</span>
   521→              )}
   522→              <span className="text-gray-600 flex-1 truncate">{part.partName}</span>
   523→              <Badge variant="secondary" className="text-xs h-5 px-2">{t('photo.quantity')}: {part.quantity}</Badge>
   524→              {getCriticalityBadge(part.criticality)}
   525→              <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700 min-h-[44px] min-w-[44px] md:min-h-0 md:min-w-0" onClick={() => onRemovePart(part.id)}>
   526→                <X className="h-4 w-4" />
   527→              </Button>
   528→            </div>
   529→          ))}
   530→        </div>
   531→      )}
   532→
   533→      {/* Mobile layout - stacked */}
   534→      <div className="md:hidden space-y-2">
   535→        <div className="grid grid-cols-2 gap-2">
   536→          <Input className="h-11 text-base" placeholder="PN" value={newPn} onChange={(e) => setNewPn(e.target.value)} />
   537→          <Input className="h-11 text-base" placeholder={t('photo.serialPlaceholder')} value={newSerialNumber} onChange={(e) => setNewSerialNumber(e.target.value)} />
   538→        </div>
   539→        <Input className="h-11 text-base" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   540→        <div className="grid grid-cols-2 gap-2">
   541→          <Input className="h-11 text-base" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />
   542→          <Select value={newCriticality} onValueChange={(v) => setNewCriticality(v as 'Alta' | 'Média' | 'Baixa' | '')}>
   543→            <SelectTrigger className="h-11 text-base">
   544→              <SelectValue placeholder={t('photo.criticality')} />
   545→            </SelectTrigger>
   546→            <SelectContent>
   547→              <SelectItem value="Alta">{t('photo.criticalityHigh')}</SelectItem>
   548→              <SelectItem value="Média">{t('photo.criticalityMedium')}</SelectItem>
   549→              <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>

   559→      <div className="hidden md:grid grid-cols-12 gap-2">

   562→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.partNamePlaceholder')} value={newPartName} onChange={(e) => setNewPartName(e.target.value)} />
   563→        <Input className="h-8 text-sm col-span-2" placeholder={t('photo.quantityPlaceholder')} type="number" value={newQuantity} onChange={(e) => setNewQuantity(e.target.value)} />
   564→        <Select value={newCriticality} onValueChange={(v) => setNewCriticality(v as 'Alta' | 'Média' | 'Baixa' | '')}>
   565→          <SelectTrigger className="h-8 text-xs col-span-2">
   566→            <SelectValue placeholder={t('photo.criticality')} />
   567→          </SelectTrigger>
   568→          <SelectContent>
   569→            <SelectItem value="Alta">{t('photo.criticalityHigh')}</SelectItem>
   570→            <SelectItem value="Média">{t('photo.criticalityMedium')}</SelectItem>
   571→            <SelectItem value="Baixa">{t('photo.criticalityLow')}</SelectItem>
   572→          </SelectContent>
   573→        </Select>
   574→        <Button size="icon" className="h-8 w-8 bg-orange-500 hover:bg-orange-600 col-span-2" onClick={handleAddPart} disabled={!newPn.trim() || !newPartName.trim() || !newQuantity.trim()}>
   575→          <Plus className="h-4 w-4" />
   576→        </Button>
   577→      </div>
   578→        </>
   579→      )}
   580→    </div>
   581→  );
   582→}

   587→// Editor de Foto
   588→function PhotoEditor({
   589→  photo,
   590→  editMode = 'primary',
   591→  onSave,
   592→  onClose,
   593→  t,
   594→}: {
   595→  photo: PhotoData;
   596→  editMode?: PhotoEditMode;
   597→  onSave: (data: Partial<PhotoData>) => void;
   598→  onClose: () => void;
   599→  t: (key: string, params?: Record<string, string | number>) => string;
   600→}) {
   601→  const canvasRef = useRef<HTMLCanvasElement>(null);
   602→  const containerRef = useRef<HTMLDivElement>(null);
   603→  const imgRef = useRef<HTMLImageElement | null>(null);
   604→  
   605→  const [objects, setObjects] = useState<EditorCircle[]>(() => {
   606→    return (photo.embeddedPhotos || []).map(ep => ({
   607→      id: ep.id,
   608→      type: 'circleWithPhoto' as const,
   609→      x: ep.circleX,
   610→      y: ep.circleY,
   611→      radius: ep.circleRadius,
   612→      color: '#f97316',
   613→      imageData: ep.imageData,

   615→  });
   616→

   618→  const [selectedObjectId, setSelectedObjectId] = useState<string | null>(null);
   619→  const [isDragging, setIsDragging] = useState(false);
   620→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 });
   621→  const [currentColor, setCurrentColor] = useState('#f97316');
   622→  const [circleRadius, setCircleRadius] = useState(80);
   623→  const [showAddPhotoDialog, setShowAddPhotoDialog] = useState(false);
   624→  const [pendingPhotoCircle, setPendingPhotoCircle] = useState<EditorCircle | null>(null);
   625→  const [isDrawingArrow, setIsDrawingArrow] = useState(false);
   626→  const [arrowStart, setArrowStart] = useState<{ x: number; y: number } | null>(null);
   627→  const [arrowPreview, setArrowPreview] = useState<{ x: number; y: number } | null>(null);
   628→  const [isDrawingCircle, setIsDrawingCircle] = useState(false);
   629→  const [circleStart, setCircleStart] = useState<{ x: number; y: number } | null>(null);
   630→  const [circlePreview, setCirclePreview] = useState<{ x: number; y: number; radius: number } | null>(null);
   631→  const [isDrawingRect, setIsDrawingRect] = useState(false);
   632→  const [rectStart, setRectStart] = useState<{ x: number; y: number } | null>(null);
   633→  const [rectPreview, setRectPreview] = useState<{ x: number; y: number; w: number; h: number } | null>(null);
   634→  const [rotation, setRotation] = useState(0);
   635→
   636→  // Determinar qual imagem editar baseado no modo
   637→  const getActiveImageSrc = () => {
   638→    if (editMode === 'secondary') {
   639→      return photo.secondaryImageData;
   640→    }
   641→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   642→      // Criar imagem combinada lado a lado
   643→      return null; // Será tratado separadamente
   644→    }
   645→    return photo.editedImageData || photo.imageData;
   646→  };
   647→  
   648→  const imageSrc = getActiveImageSrc();
   649→  const [dualCanvasSize, setDualCanvasSize] = useState<{ width: number; height: number } | null>(null);
   650→  const [canvasSize, setCanvasSize] = useState({ width: 0, height: 0, scale: 1 });
   651→
   652→  useEffect(() => {
   653→    const container = containerRef.current;
   654→    if (!container) return;
   655→    const containerWidth = Math.min(container.clientWidth - 32, 900);
   656→    const containerHeight = 500;
   657→
   658→    // Modo duplo: carregar ambas as imagens lado a lado
   659→    if (editMode === 'dual' && photo.imageData && photo.secondaryImageData) {
   660→      const img1 = new window.Image();
   661→      const img2 = new window.Image();
   662→      
   663→      let loaded = 0;
   664→      const checkLoaded = () => {
   665→        loaded++;
   666→        if (loaded === 2) {
   667→          // Calcular dimensões para ambas as imagens lado a lado
   668→          const halfWidth = containerWidth / 2 - 10;
   669→          const scale1 = Math.min(halfWidth / img1.width, containerHeight / img1.height);
   670→          const scale2 = Math.min(halfWidth / img2.width, containerHeight / img2.height);
   671→          const scale = Math.min(scale1, scale2);
   672→          
   673→          const totalWidth = (img1.width + img2.width) * scale + 10;
   674→          const totalHeight = Math.max(img1.height, img2.height) * scale;
   675→          
   676→          setCanvasSize({ width: totalWidth, height: totalHeight, scale });
   677→          setDualCanvasSize({ width: img1.width * scale, height: img1.height * scale });
   678→          
   679→          // Criar imagem combinada temporária
   680→          const tempCanvas = document.createElement('canvas');
   681→          tempCanvas.width = img1.width + img2.width;
   682→          tempCanvas.height = Math.max(img1.height, img2.height);
   683→          const ctx = tempCanvas.getContext('2d');
   684→          if (ctx) {
   685→            ctx.drawImage(img1, 0, 0);
   686→            ctx.drawImage(img2, img1.width, 0);
   687→          }
   688→          
   689→          const combinedImg = new window.Image();
   690→          combinedImg.onload = () => {
   691→            imgRef.current = combinedImg;
   692→          };
   693→          combinedImg.src = tempCanvas.toDataURL('image/png');
   694→        }
   695→      };
   696→      
   697→      img1.onload = checkLoaded;
   698→      img2.onload = checkLoaded;
   699→      img1.src = photo.imageData;
   700→      img2.src = photo.secondaryImageData;
   701→      return;
   702→    }
   703→
   704→    // Modo normal (primário ou secundário)
   705→    if (!imageSrc) return;
   706→    const img = new window.Image();
   707→    img.onload = () => {
   708→      imgRef.current = img;
   709→      let imgWidth = img.width;
   710→      let imgHeight = img.height;
   711→      // Swap dimensions for 90/270 degree rotation
   712→      if (rotation === 90 || rotation === 270) {
   713→        [imgWidth, imgHeight] = [imgHeight, imgWidth];
   714→      }
   715→      const scale = Math.min(containerWidth / imgWidth, containerHeight / imgHeight);
   716→      setCanvasSize({ width: imgWidth * scale, height: imgHeight * scale, scale });
   717→    };
   718→    img.src = imageSrc;
   719→  }, [imageSrc, editMode, photo.imageData, photo.secondaryImageData, rotation]);
   720→
   721→  const drawCanvas = useCallback(() => {
   722→    const canvas = canvasRef.current;
   723→    const img = imgRef.current;
   724→    if (!canvas || !img || !imageSrc) return;
   725→
   726→    const ctx = canvas.getContext('2d');
   727→    if (!ctx) return;
   728→
   729→    const { width, height, scale } = canvasSize;
   730→    canvas.width = width;
   731→    canvas.height = height;
   732→    ctx.save();
   733→    if (rotation > 0) {
   734→      ctx.translate(width / 2, height / 2);
   735→      ctx.rotate((rotation * Math.PI) / 180);
   736→      ctx.translate(-width / 2, -height / 2);
   737→    }
   738→    ctx.drawImage(img, 0, 0, width, height);
   739→    ctx.restore();
   740→
   741→    objects.forEach((obj) => {
   742→      const isSelected = obj.id === selectedObjectId;
   743→      
   744→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   745→        const x = obj.x * scale;
   746→        const y = obj.y * scale;
   747→        const radius = (obj.radius || 50) * scale;
   748→
   749→        if (obj.type === 'circleWithPhoto' && obj.imageData) {
   750→          const embeddedImg = new window.Image();
   751→          embeddedImg.onload = () => {
   752→            ctx.save();
   753→            ctx.beginPath();
   754→            ctx.arc(x, y, radius - 2, 0, Math.PI * 2);
   755→            ctx.closePath();
   756→            ctx.clip();
   757→            ctx.drawImage(embeddedImg, x - radius + 2, y - radius + 2, (radius - 2) * 2, (radius - 2) * 2);
   758→            ctx.restore();
   759→            ctx.beginPath();
   760→            ctx.arc(x, y, radius, 0, Math.PI * 2);
   761→            ctx.strokeStyle = obj.color;
   762→            ctx.lineWidth = isSelected ? 4 : 3;
   763→            ctx.stroke();
   764→          };
   765→          embeddedImg.src = obj.imageData;
   766→        } else {
   767→          ctx.beginPath();
   768→          ctx.arc(x, y, radius, 0, Math.PI * 2);
   769→          ctx.strokeStyle = obj.color;
   770→          ctx.lineWidth = isSelected ? 4 : 3;
   771→          ctx.stroke();
   772→        }
   773→      }
   774→      
   775→      if (obj.type === 'arrow') {
   776→        const startX = obj.x * scale;
   777→        const startY = obj.y * scale;
   778→        const endX = (obj.endX || obj.x + 50) * scale;
   779→        const endY = (obj.endY || obj.y + 50) * scale;
   780→
   781→        ctx.beginPath();
   782→        ctx.moveTo(startX, startY);
   783→        ctx.lineTo(endX, endY);
   784→        ctx.strokeStyle = obj.color;
   785→        ctx.lineWidth = isSelected ? 4 : 3;
   786→        ctx.stroke();
   787→
   788→        const angle = Math.atan2(endY - startY, endX - startX);
   789→        ctx.beginPath();
   790→        ctx.moveTo(endX, endY);
   791→        ctx.lineTo(endX - 15 * Math.cos(angle - Math.PI / 6), endY - 15 * Math.sin(angle - Math.PI / 6));
   792→        ctx.moveTo(endX, endY);
   793→        ctx.lineTo(endX - 15 * Math.cos(angle + Math.PI / 6), endY - 15 * Math.sin(angle + Math.PI / 6));
   794→        ctx.stroke();
   795→      }
   796→    });
   797→
   798→    // Rectangle objects
   799→    objects.forEach((obj) => {
   800→      if (obj.type === 'rectangle') {
   801→        const isSelected = obj.id === selectedObjectId;
   802→        const rx = obj.x * scale;
   803→        const ry = obj.y * scale;
   804→        const rw = (obj.width || 100) * scale;
   805→        const rh = (obj.height || 80) * scale;
   806→        ctx.strokeStyle = obj.color;
   807→        ctx.lineWidth = isSelected ? 4 : 3;
   808→        ctx.strokeRect(rx, ry, rw, rh);
   809→      }
   810→    });
   811→
   812→    if (isDrawingArrow && arrowStart && arrowPreview) {
   813→      ctx.beginPath();
   814→      ctx.moveTo(arrowStart.x * scale, arrowStart.y * scale);
   815→      ctx.lineTo(arrowPreview.x * scale, arrowPreview.y * scale);
   816→      ctx.strokeStyle = currentColor;
   817→      ctx.lineWidth = 3;
   818→      ctx.setLineDash([5, 5]);
   819→      ctx.stroke();
   820→      ctx.setLineDash([]);
   821→    }
   822→
   823→    // Circle drag preview
   824→    if (isDrawingCircle && circleStart && circlePreview) {
   825→      ctx.beginPath();
   826→      ctx.arc(circlePreview.x * scale, circlePreview.y * scale, circlePreview.radius * scale, 0, Math.PI * 2);
   827→      ctx.strokeStyle = currentColor;
   828→      ctx.lineWidth = 3;
   829→      ctx.setLineDash([5, 5]);
   830→      ctx.stroke();
   831→      ctx.setLineDash([]);
   832→    }
   833→
   834→    // Rectangle drag preview
   835→    if (isDrawingRect && rectStart && rectPreview) {
   836→      ctx.strokeStyle = currentColor;
   837→      ctx.lineWidth = 3;
   838→      ctx.setLineDash([5, 5]);
   839→      ctx.strokeRect(rectPreview.x * scale, rectPreview.y * scale, rectPreview.w * scale, rectPreview.h * scale);
   840→      ctx.setLineDash([]);
   841→    }
   842→  }, [imageSrc, canvasSize, objects, selectedObjectId, isDrawingArrow, arrowStart, arrowPreview, currentColor, rotation, isDrawingCircle, circleStart, circlePreview, isDrawingRect, rectStart, rectPreview]);
   843→
   844→  useEffect(() => {
   845→    drawCanvas();
   846→  }, [drawCanvas]);
   847→
   848→  const getMousePos = (e: React.MouseEvent): { x: number; y: number } => {
   849→    const canvas = canvasRef.current;
   850→    if (!canvas) return { x: 0, y: 0 };
   851→    const rect = canvas.getBoundingClientRect();
   852→    return { x: (e.clientX - rect.left) / canvasSize.scale, y: (e.clientY - rect.top) / canvasSize.scale };
   853→  };
   854→
   855→  // Touch position helper for mobile
   856→  const getTouchPos = (e: React.TouchEvent): { x: number; y: number } => {
   857→    const canvas = canvasRef.current;
   858→    if (!canvas || e.touches.length === 0) return { x: 0, y: 0 };
   859→    const rect = canvas.getBoundingClientRect();
   860→    const touch = e.touches[0];
   861→    return { x: (touch.clientX - rect.left) / canvasSize.scale, y: (touch.clientY - rect.top) / canvasSize.scale };
   862→  };
   863→
   864→  const findObjectAtPosition = (x: number, y: number): EditorCircle | null => {
   865→    for (let i = objects.length - 1; i >= 0; i--) {
   866→      const obj = objects[i];
   867→      if (obj.type === 'circle' || obj.type === 'circleWithPhoto') {
   868→        const dist = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   869→        if (dist <= (obj.radius || 50)) return obj;
   870→      }
   871→      if (obj.type === 'arrow') {
   872→        const lineLength = Math.sqrt((obj.endX! - obj.x) ** 2 + (obj.endY! - obj.y) ** 2);
   873→        const d1 = Math.sqrt((x - obj.x) ** 2 + (y - obj.y) ** 2);
   874→        const d2 = Math.sqrt((x - obj.endX!) ** 2 + (y - obj.endY!) ** 2);
   875→        if (d1 + d2 <= lineLength + 15) return obj;
   876→      }
   877→      if (obj.type === 'rectangle') {
   878→        const rx = Math.min(obj.x, obj.x + (obj.width || 100));
   879→        const ry = Math.min(obj.y, obj.y + (obj.height || 80));
   880→        const rw = Math.abs(obj.width || 100);
   881→        const rh = Math.abs(obj.height || 80);
   882→        if (x >= rx - 10 && x <= rx + rw + 10 && y >= ry - 10 && y <= ry + rh + 10) return obj;
   883→      }
   884→    }
   885→    return null;
   886→  };
   887→
   888→  const handleMouseDown = (e: React.MouseEvent) => {
   889→    const pos = getMousePos(e);
   890→    if (currentTool === 'select') {
   891→      const obj = findObjectAtPosition(pos.x, pos.y);
   892→      if (obj) {
   893→        setSelectedObjectId(obj.id);
   894→        setIsDragging(true);
   895→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   896→      } else {
   897→        setSelectedObjectId(null);
   898→      }
   899→    } else if (currentTool === 'circle') {
   900→      setIsDrawingCircle(true);
   901→      setCircleStart(pos);
   902→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
   903→    } else if (currentTool === 'circlePhoto') {
   904→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
   905→      setShowAddPhotoDialog(true);
   906→    } else if (currentTool === 'arrow') {
   907→      setIsDrawingArrow(true);
   908→      setArrowStart(pos);
   909→      setArrowPreview(pos);
   910→    } else if (currentTool === 'rectangle') {
   911→      setIsDrawingRect(true);
   912→      setRectStart(pos);
   913→      setRectPreview({ x: pos.x, y: pos.y, w: 1, h: 1 });
   914→    }
   915→  };
   916→
   917→  const handleMouseMove = (e: React.MouseEvent) => {
   918→    const pos = getMousePos(e);
   919→    if (isDragging && selectedObjectId) {
   920→      setObjects(prev => prev.map(obj => {
   921→        if (obj.id !== selectedObjectId) return obj;
   922→        const newX = pos.x - dragOffset.x;
   923→        const newY = pos.y - dragOffset.y;
   924→        if (obj.type === 'arrow') {
   925→          const dx = obj.endX! - obj.x;
   926→          const dy = obj.endY! - obj.y;
   927→          return { ...obj, x: newX, y: newY, endX: newX + dx, endY: newY + dy };
   928→        }
   929→        if (obj.type === 'rectangle') {
   930→          return { ...obj, x: newX, y: newY };
   931→        }
   932→        return { ...obj, x: newX, y: newY };
   933→      }));
   934→    }
   935→    if (isDrawingArrow) setArrowPreview(pos);
   936→    if (isDrawingCircle && circleStart) {
   937→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   938→      setCirclePreview({ x: circleStart.x, y: circleStart.y, radius: dist });
   939→    }
   940→    if (isDrawingRect && rectStart) {
   941→      setRectPreview({ x: rectStart.x, y: rectStart.y, w: pos.x - rectStart.x, h: pos.y - rectStart.y });
   942→    }
   943→  };
   944→
   945→  const handleMouseUp = (e: React.MouseEvent) => {
   946→    const pos = getMousePos(e);
   947→    if (isDragging) setIsDragging(false);
   948→    if (isDrawingCircle && circleStart && currentTool === 'circle') {
   949→      const dist = Math.sqrt((pos.x - circleStart.x) ** 2 + (pos.y - circleStart.y) ** 2);
   950→      const finalRadius = dist > 10 ? dist : circleRadius;
   951→      const newCircle: EditorCircle = { id: `circle-${Date.now()}`, type: 'circle', x: circleStart.x, y: circleStart.y, radius: finalRadius, color: currentColor };
   952→      setObjects(prev => [...prev, newCircle]);
   953→      setSelectedObjectId(newCircle.id);
   954→      setIsDrawingCircle(false);
   955→      setCircleStart(null);
   956→      setCirclePreview(null);
   957→      setCurrentTool('select');
   958→    }
   959→    if (isDrawingRect && rectStart && currentTool === 'rectangle') {
   960→      const w = pos.x - rectStart.x;
   961→      const h = pos.y - rectStart.y;
   962→      if (Math.abs(w) > 10 && Math.abs(h) > 10) {
   963→        const newRect: EditorCircle = { id: `rect-${Date.now()}`, type: 'rectangle', x: rectStart.x, y: rectStart.y, width: w, height: h, color: currentColor };
   964→        setObjects(prev => [...prev, newRect]);
   965→        setSelectedObjectId(newRect.id);
   966→      }
   967→      setIsDrawingRect(false);
   968→      setRectStart(null);
   969→      setRectPreview(null);
   970→      setCurrentTool('select');
   971→    }
   972→    if (isDrawingArrow && arrowStart && currentTool === 'arrow') {
   973→      const distance = Math.sqrt((pos.x - arrowStart.x) ** 2 + (pos.y - arrowStart.y) ** 2);
   974→      if (distance > 10) {
   975→        const newArrow: EditorCircle = { id: `arrow-${Date.now()}`, type: 'arrow', x: arrowStart.x, y: arrowStart.y, endX: pos.x, endY: pos.y, color: currentColor };
   976→        setObjects(prev => [...prev, newArrow]);
   977→        setSelectedObjectId(newArrow.id);
   978→      }
   979→      setIsDrawingArrow(false);
   980→      setArrowStart(null);
   981→      setArrowPreview(null);
   982→      setCurrentTool('select');
   983→    }
   984→  };
   985→
   986→  // Touch event handlers for mobile
   987→  const handleTouchStart = (e: React.TouchEvent) => {
   988→    e.preventDefault();
   989→    const pos = getTouchPos(e);
   990→    if (currentTool === 'select') {
   991→      const obj = findObjectAtPosition(pos.x, pos.y);
   992→      if (obj) {
   993→        setSelectedObjectId(obj.id);
   994→        setIsDragging(true);
   995→        setDragOffset({ x: pos.x - obj.x, y: pos.y - obj.y });
   996→      } else {
   997→        setSelectedObjectId(null);
   998→      }
   999→    } else if (currentTool === 'circle') {
  1000→      setIsDrawingCircle(true);
  1001→      setCircleStart(pos);
  1002→      setCirclePreview({ x: pos.x, y: pos.y, radius: 5 });
  1003→    } else if (currentTool === 'circlePhoto') {
  1004→      setPendingPhotoCircle({ id: `cp-${Date.now()}`, type: 'circleWithPhoto', x: pos.x, y: pos.y, radius: circleRadius, color: currentColor });
  1005→      setShowAddPhotoDialog(true);
  1006→    } else if (currentTool === 'arrow') {
  1007→      setIsDrawingArrow(true);
  1008→      setArrowStart(pos);
  1009→      setArrowPreview(pos);
  1010→    } else if (currentTool === 'rectangle') {

  1303→// Função para gerar imagem preta com número de série
  1304→function generateSerialNumberImage(serialNumber: string | undefined): string | null {
  1305→  if (!serialNumber || serialNumber.trim() === '') return null;
  1306→  
  1307→  const canvas = document.createElement('canvas');
  1308→  canvas.width = 400;
  1309→  canvas.height = 200;
  1310→  const ctx = canvas.getContext('2d');
  1311→  
  1312→  if (!ctx) return null;
  1313→  
  1314→  // Fundo preto
  1315→  ctx.fillStyle = '#000000';
  1316→  ctx.fillRect(0, 0, canvas.width, canvas.height);
  1317→  
  1318→  // Texto branco
  1319→  ctx.fillStyle = '#FFFFFF';
  1320→  ctx.font = 'bold 32px monospace';
  1321→  ctx.textAlign = 'center';
  1322→  ctx.textBaseline = 'middle';
  1323→  
  1324→  // Desenhar o número de série
  1325→  ctx.fillText(serialNumber.trim(), canvas.width / 2, canvas.height / 2);
  1326→  
  1327→  return canvas.toDataURL('image/png');
  1328→}
